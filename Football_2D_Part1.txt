  📥 КУДА ВСТАВЛЯТЬ (ДВА СКРИПТА — КЛИЕНТ + СЕРВЕР):

  ЭТОТ файл (клиент):
  1) Открой проект в Roblox Studio (подойдёт пустая Baseplate).
  2) Explorer → StarterGui → открой тот же LocalScript, что и раньше.
  3) Удали в нём весь старый код и вставь ВЕСЬ этот файл целиком.

  Файл ROBALL_Online_Server.txt (сервер онлайна):
  4) Explorer → ServerScriptService → [+] → Script (НЕ LocalScript).
  5) Удали в нём всё и вставь весь файл ROBALL_Online_Server.txt целиком.
  6) Нажми Play. RemoteEvent-ы сервер создаст сам, вручную ничего делать не надо.

  ⚠️ Один LocalScript физически не может создать комнаты, видимые другим

--// Маленькие помощники \\--
local function clamp(v, a, b)
	if v < a then return a end
	if v > b then return b end
	return v
end

local function unit2(v)
	local m = v.Magnitude
	if m < 0.0001 then
		return Vector2.new(0, 0)
	end
	return v / m
end

-- Кратчайшая плавная интерполяция угла (для поворота персонажей и повтора)
local function lerpAngle(a, b, alpha)
	local diff = (b - a + math.pi) % (math.pi * 2) - math.pi
	return a + diff * clamp(alpha, 0, 1)
end

-- Плавная физика фишки: скорость меняется не мгновенно, а с ускорением.
-- Из-за этого есть разгон, инерция, скольжение при остановке и мягкие повороты.
local function approachVelocity(current, desired, acceleration, dt)
	local delta = desired - current
	local maxChange = acceleration * dt
	if delta.Magnitude <= maxChange then
		return desired
	end
	return current + unit2(delta) * maxChange
end

-- Геометрия поля со скошенными углами: круг радиуса r не может забиться в мёртвый угол.
local function clampToField(pos, r)
	local p = Vector2.new(
		clamp(pos.X, r, FIELD_W - r),
		clamp(pos.Y, r, FIELD_H - r)
	)
	local need = CORNER_CUT + r * math.sqrt(2)
	local tl = p.X + p.Y
	if tl < need then
		local add = (need - tl) / 2
		p = Vector2.new(p.X + add, p.Y + add)
	end
	local tr = (FIELD_W - p.X) + p.Y
	if tr < need then
		local add = (need - tr) / 2
		p = Vector2.new(p.X - add, p.Y + add)
	end
	local bl = p.X + (FIELD_H - p.Y)
	if bl < need then
		local add = (need - bl) / 2
		p = Vector2.new(p.X + add, p.Y - add)
	end
	local br = (FIELD_W - p.X) + (FIELD_H - p.Y)
	if br < need then
		local add = (need - br) / 2
		p = Vector2.new(p.X - add, p.Y - add)
	end
	return p
end

-- При ограничении полем убираем только скорость В стену и сохраняем скольжение вдоль неё.
-- Особенно важно на скошенных углах, где старый покоординатный сброс убивал сразу обе оси.
local function applyFieldClamp(pos, vel, r)
	local clamped = clampToField(pos, r)
	local correction = clamped - pos
	if correction.Magnitude > 0.0001 then
		local inward = unit2(correction)
		local intoWallSpeed = vel:Dot(inward)
		if intoWallSpeed < 0 then
			vel = vel - inward * intoWallSpeed
		end
	end
	return clamped, vel
end

local function isInCornerZone(pos, pad)
	local scaledPad = pad * FIELD_SCALE -- угловая зона растёт вместе с большим полем
	return (pos.X < scaledPad or pos.X > FIELD_W - scaledPad)
		and (pos.Y < scaledPad or pos.Y > FIELD_H - scaledPad)
end

-- Ближайшая стена и направление ВНУТРЬ поля. Это общая защита для всех ботов.
local function getNearestWall(pos)
	local left = pos.X
	local right = FIELD_W - pos.X
	local top = pos.Y
	local bottom = FIELD_H - pos.Y
	local best = left
	local inward = Vector2.new(1, 0)
	local tangent = Vector2.new(0, 1)
	local side = "Left"
	if right < best then
		best = right
		inward = Vector2.new(-1, 0)
		tangent = Vector2.new(0, 1)
		side = "Right"
	end
	if top < best then
		best = top
		inward = Vector2.new(0, 1)
		tangent = Vector2.new(1, 0)
		side = "Top"
	end
	if bottom < best then
		best = bottom
		inward = Vector2.new(0, -1)
		tangent = Vector2.new(1, 0)
		side = "Bottom"
	end
	return best, inward, tangent, side
end

local function clampBotTarget(pos)
	return clampToField(pos, BOT_TARGET_MARGIN)
end

local function getWallBallInfo(pos)
	local dist, inward, tangent, side = getNearestWall(pos)
	local inGoalMouth = (side == "Left" or side == "Right")
		and pos.Y > GOAL_Y1 + BALL_R and pos.Y < GOAL_Y2 - BALL_R
	return (dist <= WALL_BALL_ZONE and not inGoalMouth), dist, inward, tangent, side
end

-- Обычный заход за мяч может оказаться за стенкой. Тогда он физически недостижим:
-- фишка не может встать по ту сторону мяча, даже если цель потом обрезать clamp-ом.
local function getReachableApproach(ballPosition, desiredDir)
	if desiredDir.Magnitude < 0.01 then
		return false, clampBotTarget(ballPosition)
	end
	local raw = ballPosition - desiredDir * (PLAYER_R + BALL_R + 8)
	local safe = clampBotTarget(raw)
	if (raw - safe).Magnitude > 10 then
		return false, safe
	end
	local contactDir = unit2(ballPosition - safe)
	return contactDir:Dot(desiredDir) >= 0.18, safe
end

-- У стены мяч нельзя выбить строго внутрь с невозможной точки за ним.
-- Бот заходит сбоку вдоль борта: касание отправляет мяч вдоль стены,
-- а небольшой компонент в стену сразу отражается обратно в поле.
local function getWallExtraction(ballPosition, attackDirX, preferredTarget, fromPos)
	local isWall, _, inward, rawTangent = getWallBallInfo(ballPosition)
	if not isWall then
		return false
	end
	local tangent
	if math.abs(rawTangent.X) > 0.5 then
		local tx = attackDirX or 1
		-- Главное — реально дойти до мяча: заход выбирается с той стороны вдоль стены,
		-- где бот уже находится. Иначе он начинал обходить мяч и выглядел избегающим.
		if fromPos and math.abs(fromPos.X - ballPosition.X) >= 18 then
			tx = (fromPos.X <= ballPosition.X) and 1 or -1
		end
		if ballPosition.X < 170 * FIELD_SCALE then
			tx = 1
		elseif ballPosition.X > FIELD_W - 170 * FIELD_SCALE then
			tx = -1
		end
		tangent = Vector2.new(tx, 0)
	else
		local ty
		if fromPos and math.abs(fromPos.Y - ballPosition.Y) >= 18 then
			ty = (fromPos.Y <= ballPosition.Y) and 1 or -1
		elseif preferredTarget and math.abs(preferredTarget.Y - ballPosition.Y) > 45 then
			ty = (preferredTarget.Y > ballPosition.Y) and 1 or -1
		else
			ty = (ballPosition.Y < GOAL_CENTER_Y) and 1 or -1
		end
		if ballPosition.Y < 170 * FIELD_SCALE then
			ty = 1
		elseif ballPosition.Y > FIELD_H - 170 * FIELD_SCALE then
			ty = -1
		end
		tangent = Vector2.new(0, ty)
	end
	local extractionDir = unit2(tangent + inward * 0.42)
	local aim = clampToField(ballPosition + tangent * (270 * FIELD_SCALE) + inward * (145 * FIELD_SCALE), BALL_R)
	local approach = clampToField(ballPosition - tangent * 34 + inward * 25, PLAYER_R + 4)
	return true, approach, aim, extractionDir, tangent, inward
end

-- Если мяч в мёртвой точке у стены/угла и физически никто не может нормально
-- зайти за него, мяч сам получает честный отскок в открытую зону.
local function getDeadBallEscapeDirection(pos)
	local isWall, _, inward, rawTangent = getWallBallInfo(pos)
	if not isWall then
		return nil
	end
	if isInCornerZone(pos, 125) then
		return unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - pos)
	end
	local openTangent
	if math.abs(rawTangent.X) > 0.5 then
		openTangent = Vector2.new((pos.X < FIELD_W / 2) and 1 or -1, 0)
	else
		openTangent = Vector2.new(0, (pos.Y < GOAL_CENTER_Y) and 1 or -1)
	end
	local function makeDirection(tangentDir)
		return unit2(inward * 1.12 + tangentDir * 0.52)
	end
	local dir = makeDirection(openTangent)
	local probe = pos + dir * 74
	local blocked = (probe - playerPos).Magnitude < 48 or (probe - botPos).Magnitude < 48
	for _, actor in ipairs(extraActors) do
		if actor.Active and (probe - actor.Pos).Magnitude < 48 then
			blocked = true
			break
		end
	end
	if blocked then
		dir = makeDirection(-openTangent)
	end
	return dir
end

-- Мягкая подушка у борта для обычной навигации ботов: скорость В стену
-- заранее уменьшается до нуля у безопасной границы. Финальный контакт с мячом у стены разрешён отдельно.
local function limitBotVelocityAtWall(pos, desired, allowContact)
	if allowContact or desired.Magnitude < 0.01 then
		return desired
	end
	local wallDist, inward = getNearestWall(pos)
	if wallDist >= BOT_TARGET_MARGIN + 33 then
		return desired
	end
	local inwardSpeed = desired:Dot(inward) -- отрицательная = движение к стене
	local allowedInwardSpeed = -math.max(0, wallDist - BOT_TARGET_MARGIN) * (4 * FIELD_SCALE)
	if inwardSpeed < allowedInwardSpeed then
		return desired + inward * (allowedInwardSpeed - inwardSpeed)
	end
	return desired
end

local function separateBotTarget(target, avoidPos, minDist)
	local offset = target - avoidPos
	local dist = offset.Magnitude
	if dist >= minDist then
		return target
	end
	local dir
	if dist > 0.01 then
		dir = offset / dist
	else
		dir = Vector2.new(0, (target.Y < GOAL_CENTER_Y) and 1 or -1)
	end
	return clampBotTarget(avoidPos + dir * minDist)
end

local function isScreenPosInsideField(screenPos)
	-- В крупном плане поле увеличено и частично за кадром: удар принимаем по всей
	-- игровой зоне ниже верхней панели (сама панель с кнопками не трогается).

		local gs = UI.gameFrame.AbsoluteSize
		return screenPos.Y > gs.Y * 0.108 and screenPos.Y < gs.Y * 0.99
	end
	local ap = UI.fieldFrame.AbsolutePosition
	local as = UI.fieldFrame.AbsoluteSize
	return screenPos.X >= ap.X and screenPos.X <= ap.X + as.X
		and screenPos.Y >= ap.Y and screenPos.Y <= ap.Y + as.Y
end

local function refreshMenuSummary()
	local enemyName = DIFFICULTIES[selectedEnemyDiffKey].ButtonText
	local allyName = DIFFICULTIES[selectedAllyDiffKey].ButtonText
	local cameraName = (cameraMode == "Full") and "всё поле" or "только я"
	local selectedTeamSize = (matchMode == "3v3" and 3) or (matchMode == "2v2" and 2) or 1
	local fieldInfo = FIELD_SIZES[selectedTeamSize]
	local fieldText = "поле " .. tostring(fieldInfo.W) .. "×" .. tostring(fieldInfo.H) .. " (" .. fieldInfo.Label .. ")"
	if modeCardNote then
		modeCardNote.Text = "Выбрано: " .. fieldText
	end
	if UI.menuSummaryLabel then
		if matchMode == "3v3" then
			UI.menuSummaryLabel.Text = "3×3  •  🔴 " .. enemyName .. "  •  🔵 " .. allyName .. "  •  " .. fieldText .. "  •  " .. cameraName
		elseif matchMode == "2v2" then
			UI.menuSummaryLabel.Text = "2×2  •  🔴 " .. enemyName .. "  •  🔵 " .. allyName .. "  •  " .. fieldText .. "  •  " .. cameraName
		else
			UI.menuSummaryLabel.Text = "1×1  •  🔴 " .. enemyName .. "  •  " .. fieldText .. "  •  " .. cameraName
		end
	end
	if playButton then
		if matchMode == "3v3" then
			playButton.Text = "▶  ИГРАТЬ 3 НА 3"
		elseif matchMode == "2v2" then
			playButton.Text = "▶  ИГРАТЬ 2 НА 2"
		else
			playButton.Text = "▶  ИГРАТЬ 1 НА 1"
		end
	end
	if enemyCardNote then
		if matchMode == "3v3" then
			enemyCardNote.Text = "Сложность трёх соперников"
		elseif matchMode == "2v2" then
			enemyCardNote.Text = "Сложность двух соперников"
		else
			enemyCardNote.Text = "Один бот-соперник"
		end
	end
	if allyCardNote then
		if matchMode == "3v3" then
			allyCardNote.Text = "Два напарника за синих"
		elseif matchMode == "2v2" then
			allyCardNote.Text = "Один напарник за синих"
		else
			allyCardNote.Text = "Появятся в режимах 2×2 и 3×3"
		end
	end
end

local function refreshModeButtons()
	for key, info in pairs(UI.modeButtons) do
		if key == matchMode then
			info.Button.BackgroundColor3 = COL.GreenBtn
			info.Stroke.Transparency = 0
		else
			info.Button.BackgroundColor3 = COL.GreyBtn
			info.Stroke.Transparency = 0.78
		end
	end
	-- В 1 на 1 карточка напарника остаётся в каталоге, но выглядит приглушённой.
	for _, info in pairs(UI.allyDiffButtons) do
		info.Button.BackgroundTransparency = (matchMode ~= "1v1") and 0 or 0.28
		info.Button.TextTransparency = (matchMode ~= "1v1") and 0 or 0.38
	end
	refreshMenuSummary()
end

local function refreshBotDifficultyButtons()
	for key, info in pairs(UI.enemyDiffButtons) do
		if key == selectedEnemyDiffKey then
			info.Button.BackgroundColor3 = DIFFICULTIES[key].Color
			info.Stroke.Transparency = 0
		else
			info.Button.BackgroundColor3 = COL.GreyBtn
			info.Stroke.Transparency = 0.78
		end

	for key, info in pairs(UI.allyDiffButtons) do

			info.Button.BackgroundColor3 = DIFFICULTIES[key].Color
			info.Stroke.Transparency = 0
		else
			info.Button.BackgroundColor3 = COL.GreyBtn
			info.Stroke.Transparency = 0.78
		end
	end
	refreshMenuSummary()
end

local function avatarImageFor(userId)
	if type(userId) == "number" and userId > 0 then
		return "rbxthumb://type=AvatarHeadShot&id=" .. tostring(math.floor(userId)) .. "&w=420&h=420"
	end
	return ""
end

local function makeAvatar(parent, x, y, sizePx, userId, displayName, team)
	local holder = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(x, 0, y, 0),
		Size = UDim2.fromOffset(sizePx, sizePx),
		BackgroundColor3 = (team == "Red") and COL.Bot or COL.Player,
		BorderSizePixel = 0,
		ZIndex = 8,
	}, parent)
	addCorner(holder, 1, 0)
	addStroke(holder, COL.White, 2, 0.15)
	local initial = "?"
	if displayName and displayName ~= "" then
		initial = string.sub(displayName, 1, 1)
	end
	makeText(holder, {
		Size = UDim2.fromScale(1, 1),
		Text = initial,
		Font = Enum.Font.GothamBlack,
		TextSize = 22,
		TextScaled = true,
		TextColor3 = COL.White,
		ZIndex = 8,
	})
	local image = avatarImageFor(userId)
	if image ~= "" then
		local img = create("ImageLabel", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 1,
			Image = image,
			ScaleType = Enum.ScaleType.Crop,
			ZIndex = 9,
		}, holder)
		addCorner(img, 1, 0)
	end
	return holder
end

local function localPlayerInRoom(room)
	if not room or type(room.Players) ~= "table" then
		return false
	end
	for _, info in ipairs(room.Players) do
		if info.UserId == player.UserId then
			return true
		end
	end
	return false
end

local function roomModeText(room)
	if room.ModeLabel then return room.ModeLabel end
	if room.Mode == "3v3" then return "3×3" end
	if room.Mode == "2v2" then return "2×2" end
	return "1×1"
end

UI.renderCurrentRoom = function()
	if not UI.onlineCurrentSlots then return end
	for _, child in ipairs(UI.onlineCurrentSlots:GetChildren()) do
		if child:IsA("GuiObject") then
			child:Destroy()
		end
	end
	local room = Net.Room
	if not room then
		for _, candidate in ipairs(Net.Rooms or {}) do
			if localPlayerInRoom(candidate) then
				room = candidate
				Net.Room = candidate
				Net.InRoom = true
				break
			end
		end
	end
	if not room then
		Net.InRoom = false
		UI.onlineCurrentMeta.Text = "Ты не в комнате • создай свою или нажми «Войти»"
		UI.onlineLeaveButton.Visible = false
		return
	end
	Net.InRoom = true
	local statusText = "Ожидание игроков"
	if room.Status == "Countdown" or room.Status == "Playing" or room.Status == "Goal" then
		statusText = "Матч идёт"
	elseif room.Status == "Finished" then
		statusText = "Матч завершён"
	end
	UI.onlineCurrentMeta.Text = string.format("Комната %s • до %d голов • %d/%d • %s",
		room.HostDisplayName or room.HostName or "игрока", room.WinGoals or 3,
		room.PlayerCount or #(room.Players or {}), room.MaxPlayers or 2, statusText)
	UI.onlineLeaveButton.Visible = true
	local bySlot = {}
	for _, info in ipairs(room.Players or {}) do
		bySlot[info.Slot] = info
	end
	local maxPlayers = room.MaxPlayers or ((room.TeamSize or 1) * 2)
	for slot = 1, maxPlayers do
		local info = bySlot[slot]
		local slotW = 0.94 / maxPlayers
		local slotFrame = create("Frame", {
			Position = UDim2.fromScale(0.03 + (slot - 1) * slotW, 0),
			Size = UDim2.fromScale(slotW - 0.012, 1),
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
			ZIndex = 7,
		}, UI.onlineCurrentSlots)
		local team = info and info.Team or ((slot % 2 == 1) and "Blue" or "Red")
		makeAvatar(slotFrame, 0.5, 0.34, 38, info and info.UserId or 0,
			info and (info.DisplayName or info.Name) or "?", team)
		makeText(slotFrame, {
			AnchorPoint = Vector2.new(0.5, 0),
			Position = UDim2.fromScale(0.5, 0.68),
			Size = UDim2.fromScale(1, 0.30),
			Text = info and (info.DisplayName or info.Name or "Игрок") or "Ожидание",
			Font = Enum.Font.GothamBold,
			TextSize = 11,
			TextScaled = true,
			TextWrapped = true,
			TextColor3 = (team == "Red") and Color3.fromRGB(255, 170, 165) or Color3.fromRGB(165, 215, 255),
			ZIndex = 8,
		})
	end
end

UI.renderOnlineRooms = function()
	if not UI.onlineRoomList then return end
	for _, child in ipairs(UI.onlineRoomList:GetChildren()) do
		if child:IsA("GuiObject") then
			child:Destroy()
		end
	end
	local rooms = Net.Rooms or {}
	UI.onlineEmptyLabel.Visible = #rooms == 0
	for i, room in ipairs(rooms) do
		local mine = localPlayerInRoom(room)
		local card = create("Frame", {
			Name = "Room_" .. tostring(room.Id or i),
			LayoutOrder = i,
			Size = UDim2.new(1, -8, 0, 78),
			BackgroundColor3 = mine and Color3.fromRGB(15, 55, 82) or Color3.fromRGB(10, 28, 46),
			BackgroundTransparency = 0.05,
			BorderSizePixel = 0,
			ZIndex = 7,
		}, UI.onlineRoomList)
		addCorner(card, 0, 14)
		addStroke(card, mine and Color3.fromRGB(80, 230, 255) or COL.White, 1, mine and 0.15 or 0.68)
		addGradient(card, Color3.fromRGB(255, 255, 255), Color3.fromRGB(120, 150, 180), 90)
		makeAvatar(card, 0, 0.5, 54, room.HostUserId, room.HostDisplayName or room.HostName, "Blue").Position = UDim2.new(0, 36, 0.5, 0)
		makeText(card, {
			Position = UDim2.new(0, 70, 0.07, 0),
			Size = UDim2.new(0.58, -70, 0.34, 0),
			Text = (room.HostDisplayName or room.HostName or "Игрок") .. ((room.HostName and room.HostName ~= room.HostDisplayName) and ("  @" .. room.HostName) or ""),
			Font = Enum.Font.GothamBlack,
			TextSize = 16,
			TextScaled = true,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextColor3 = COL.White,
			ZIndex = 8,
		})
		local statusText = "🟢 Ожидание"
		if room.Status == "Finished" then
			statusText = "🏁 Завершена"
		elseif room.Status ~= "Waiting" then
			statusText = "🔴 Идёт Матч"
		end
		makeText(card, {
			Position = UDim2.new(0, 70, 0.43, 0),
			Size = UDim2.new(0.58, -70, 0.48, 0),
			Text = string.format("%s • до %d голов • 👥 %d/%d\n%s", roomModeText(room), room.WinGoals or 3,
				room.PlayerCount or #(room.Players or {}), room.MaxPlayers or 2, statusText),
			TextSize = 13,
			TextScaled = true,
			TextWrapped = true,
			TextXAlignment = Enum.TextXAlignment.Left,
			TextColor3 = Color3.fromRGB(190, 215, 235),
			ZIndex = 8,
		})
		local canJoin = (not mine) and room.Status == "Waiting" and (room.PlayerCount or 0) < (room.MaxPlayers or 2)
		local joinText = "Войти"
		if mine then
			joinText = "Вы здесь"
		elseif room.Status ~= "Waiting" then
			joinText = "Идёт Матч"
		elseif not canJoin then
			joinText = "Полная"
		end
		local joinButton = makeButton(card, {
			AnchorPoint = Vector2.new(1, 0.5),
			Position = UDim2.fromScale(0.975, 0.5),
			Size = UDim2.fromScale(0.22, 0.58),
			BackgroundColor3 = canJoin and COL.GreenBtn or COL.GreyBtn,
			Text = joinText,
			Font = Enum.Font.GothamBlack,
			TextSize = 15,
			TextScaled = true,
			Active = canJoin,
			ZIndex = 9,
		})
		addCorner(joinButton, 0, 12)
		if canJoin then
			local roomId = room.Id
			joinButton.MouseButton1Click:Connect(function()
				if Net.JoinRoom then
					Net.JoinRoom(roomId)
				else
					UI.setOnlineStatus("Онлайн-сервер ещё не подключился")
				end
			end)
		end
	end
	UI.renderCurrentRoom()
end

	Net.MenuTab = tab
	if UI.offlinePanel then UI.offlinePanel.Visible = tab == "Offline" end
	if UI.offlinePanelShadow then UI.offlinePanelShadow.Visible = true end
	if UI.onlinePanel then UI.onlinePanel.Visible = tab == "Online" end
	UI.offlineTabButton.BackgroundColor3 = (tab == "Offline") and COL.GreenBtn or COL.GreyBtn
	UI.onlineTabButton.BackgroundColor3 = (tab == "Online") and Color3.fromRGB(0, 170, 255) or COL.GreyBtn
	if tab == "Online" then
		UI.refreshOnlineCreateButtons()
		UI.renderOnlineRooms()
		if Net.RefreshRooms then
			Net.RefreshRooms()
		end
	end
end

	local goal = create("Frame", {

local function addDisc(parent, name, cx, cy, size, color, transparency, z)
	local disc = create("Frame", {
		Name = name,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(cx, cy),
		Size = UDim2.fromScale(size, size),
		BackgroundColor3 = color,
		BackgroundTransparency = transparency or 0,
		BorderSizePixel = 0,
		ZIndex = z or 3,
	}, parent)
	addCorner(disc, 1, 0)
	return disc
end

local function makeChip(options)
	local teamColor = options.Color
	local chip = create("Frame", {
		Name = options.Name,
		AnchorPoint = Vector2.new(0.5, 0.5),
		Size = fieldSizeToUI(PLAYER_R * 2, PLAYER_R * 2),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 8,
	}, UI.fieldFrame)

	-- Только мягкая тень снаружи; внутри круга больше нет номеров и украшений.
	addDisc(chip, "Shadow", 0.56, 0.60, 0.92, COL.Black, 0.78, 1)

	local body = create("Frame", {
		Name = "Body",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromScale(1, 1),
		BackgroundColor3 = teamColor,
		BorderSizePixel = 0,
		ClipsDescendants = true,
		ZIndex = 2,
	}, chip)
	addCorner(body, 1, 0)
	addStroke(body, COL.White, 3, 0)
	addGradient(body, teamColor:Lerp(COL.White, 0.22), teamColor:Lerp(COL.Black, 0.24), 90)

	-- Чистый круглый портрет робота: тёмный визор и глаза, ничего лишнего.
	-- Рисуем его только когда нет аватара/фотки, чтобы он не выглядывал из-под картинки.
	if not options.CustomId or options.CustomId == "" then
	addDisc(body, "PortraitEarL", 0.08, 0.51, 0.17, Color3.fromRGB(190, 198, 208), 0, 3)
	addDisc(body, "PortraitEarR", 0.92, 0.51, 0.17, Color3.fromRGB(190, 198, 208), 0, 3)
	addDisc(body, "PortraitHighlight", 0.31, 0.23, 0.29, COL.White, 0.86, 3)
	local visor = create("Frame", {
		Name = "PortraitVisor",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.52),
		Size = UDim2.fromScale(0.72, 0.34),
		BackgroundColor3 = Color3.fromRGB(12, 17, 25),
		BorderSizePixel = 0,
		ZIndex = 4,
	}, body)
	addCorner(visor, 0.5, 0)
	addStroke(visor, COL.Black, 1, 0.25)
	local eyeColor = options.EyeColor or Color3.fromRGB(80, 230, 255)
	addDisc(body, "PortraitEyeL", 0.36, 0.52, 0.13, eyeColor, 0, 5)
	addDisc(body, "PortraitEyeR", 0.64, 0.52, 0.13, eyeColor, 0, 5)
	end

	-- ImageLabel создаём всегда: оффлайн здесь аватар/фото бота, онлайн сервер
	-- подставит настоящий аватар игрока, занявшего этот слот.
	local img = create("ImageLabel", {
		Name = options.PlayerAvatar and "PlayerAvatar" or "BotPhoto",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromScale(0.96, 0.96),
		BackgroundTransparency = 1,
		Image = options.CustomId or "",
		ScaleType = Enum.ScaleType.Crop,
		ZIndex = 6,
	}, body)
	addCorner(img, 1, 0)

	-- Маленький маркер направления у самого края: фото не закрывает, а играть помогает.
	local startAngle = options.InitialFacing or 0
	local marker = create("Frame", {
		Name = "DirectionMarker",
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5 + math.cos(startAngle) * 0.39, 0.5 + math.sin(startAngle) * 0.39),
		Size = UDim2.fromScale(0.15, 0.15),
		Rotation = math.deg(startAngle) + 45,
		BackgroundColor3 = options.MarkerColor or COL.Yellow,
		BorderSizePixel = 0,
		ZIndex = 9,
	}, chip)
	addCorner(marker, 0.25, 0)
	addStroke(marker, COL.Black, 1, 0.30)

	local nameLabel = makeText(chip, {
		Name = "NameLabel",
		AnchorPoint = Vector2.new(0.5, 1),
		Position = UDim2.new(0.5, 0, 0, -3),
		Size = UDim2.new(2.25, 0, 0, 18),
		Text = options.Label,
		Font = Enum.Font.GothamBlack,
		TextSize = 14,
		TextScaled = true,
		TextColor3 = COL.White,
		TextStrokeColor3 = COL.Black,
		TextStrokeTransparency = 0.30,
		ZIndex = 10,
	})
	return chip, marker, nameLabel, img
end

local function joystickCenterAndRadius()
	local base = UI.joystickBase
	local size = base.AbsoluteSize
	local center = base.AbsolutePosition + size / 2
	local knobRadius = math.max(28, math.min(size.X, size.Y) * 0.29) -- видимая ручка целиком внутри базы
	local inputRadius = math.max(44, math.min(size.X, size.Y) * 0.46) -- полный ход пальца мягче и точнее
	return center, knobRadius, size, inputRadius
end

UI.resetJoystick = function()
	UI.joystickVector = Vector2.new(0, 0)
	UI.joystickTargetVector = Vector2.new(0, 0)
	UI.joystickActive = false
	UI.joystickSprint = false
	UI.joystickInput = nil
	if UI.joystickKnob then
		UI.joystickKnob.Position = UDim2.fromScale(0.5, 0.5)
	end
end

UI.updateJoystickFromScreen = function(screenPos)
	if not UI.joystickBase then return false end
	local center, knobRadius, _, inputRadius = joystickCenterAndRadius()
	local rawOffset = screenPos - center
	local knobOffset = rawOffset
	if knobOffset.Magnitude > knobRadius then
		knobOffset = unit2(knobOffset) * knobRadius
	end
	if UI.joystickKnob then
		UI.joystickKnob.Position = UDim2.new(0.5, knobOffset.X, 0.5, knobOffset.Y)
	end

	-- Полная скорость достигается на большем ходе пальца, чем видимый ход ручки.
	-- Кривая 1.25 делает начало спокойным: телефон не срывается с места от
	-- микродвижений пальца и не получает скрытый спринт у края.
	local normalizedMagnitude = clamp(rawOffset.Magnitude / math.max(inputRadius, 1), 0, 1)
	local deadZone = 0.12
	if normalizedMagnitude <= deadZone or rawOffset.Magnitude < 0.01 then
		UI.joystickTargetVector = Vector2.new(0, 0)
	else
		local aliveMagnitude = clamp((normalizedMagnitude - deadZone) / (1 - deadZone), 0, 1)
		local curvedMagnitude = math.pow(aliveMagnitude, 1.25)
		UI.joystickTargetVector = unit2(rawOffset) * curvedMagnitude
	end
	UI.joystickSprint = false
	return true
end

UI.stepJoystick = function(dt)
	if not UI.joystickActive then
		UI.joystickVector = Vector2.new(0, 0)
		UI.joystickTargetVector = Vector2.new(0, 0)
		return
	end
	local target = UI.joystickTargetVector or Vector2.new(0, 0)
	local alpha = 1 - math.exp(-30 * math.max(dt, 0))
	UI.joystickVector = (UI.joystickVector or Vector2.new(0, 0)):Lerp(target, alpha)
	if target.Magnitude < 0.01 and UI.joystickVector.Magnitude < 0.02 then
		UI.joystickVector = Vector2.new(0, 0)
	end
	if UI.joystickVector.Magnitude > 1 then
		UI.joystickVector = unit2(UI.joystickVector)
	end
end

UI.isJoystickScreenPos = function(screenPos)
	if not UI.joystickBase then return false end
	local center, _, size = joystickCenterAndRadius()
	local captureRadius = math.min(size.X, size.Y) * 0.80
	return (screenPos - center).Magnitude <= captureRadius
end

UI.beginJoystick = function(input, screenPos)
	if not UI.joystickBase or UI.joystickActive then return false end
	UI.joystickBase.Visible = true
	if UI.kickButton then UI.kickButton.Visible = true end
	if not UI.isJoystickScreenPos(screenPos) then return false end
	UI.joystickInput = input
	UI.joystickActive = true
	UI.updateJoystickFromScreen(screenPos)
	return true
end

UI.moveJoystick = function(input, screenPos)
	if UI.joystickActive and UI.joystickInput == input then
		return UI.updateJoystickFromScreen(screenPos)
	end
	return false
end

UI.endJoystick = function(input)
	if UI.joystickInput == input then
		UI.resetJoystick()
		return true
	end
	return false
end
end -- мобильный джойстик

	end

-- Размер поля зависит от состава: 1×1 = база, 2×2 заметно шире, 3×3 — самое большое поле.
-- Важно: радиусы мяча и фишек остаются прежними в виртуальных единицах. Поэтому
-- относительно футболистов пространства действительно становится больше, а не просто
-- растягивается картинка на экране.
local function applyDynamicField()
	local cfg = FIELD_SIZES[teamSize] or FIELD_SIZES[1]
	FIELD_W = cfg.W
	FIELD_H = cfg.H
	FIELD_SCALE = cfg.W / 1000
	GOAL_CENTER_Y = FIELD_H / 2
	GOAL_Y1 = GOAL_CENTER_Y - cfg.GoalHalf
	GOAL_Y2 = GOAL_CENTER_Y + cfg.GoalHalf
	CORNER_CUT = math.floor(72 * FIELD_SCALE + 0.5)
	POSTS[1] = Vector2.new(0, GOAL_Y1)
	POSTS[2] = Vector2.new(0, GOAL_Y2)
	POSTS[3] = Vector2.new(FIELD_W, GOAL_Y1)
	POSTS[4] = Vector2.new(FIELD_W, GOAL_Y2)
	cameraCenter = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	local chipSize = fieldSizeToUI(PLAYER_R * 2, PLAYER_R * 2)
	for _, chip in ipairs({ UI.playerChip, UI.botChip, UI.allyChip, UI.ally2Chip, UI.enemy2Chip, UI.enemy3Chip }) do
		if chip then
			chip.Size = chipSize
		end
	end
	if UI.ballFrame then
		UI.ballFrame.Size = fieldSizeToUI(BALL_R * 2, BALL_R * 2)
	end
	local goalSize = fieldSizeToUI(42, GOAL_Y2 - GOAL_Y1)
	if UI.leftGoalFrame then UI.leftGoalFrame.Size = goalSize end
	if UI.rightGoalFrame then UI.rightGoalFrame.Size = goalSize end
	for i, frame in ipairs(UI.postFrames or {}) do
		frame.Position = fieldPosToUI(POSTS[i])
		frame.Size = fieldSizeToUI(16, 16)
	end
end

		playerPos = Vector2.new(FIELD_W * 0.425, FIELD_H * 0.500)
		botPos = Vector2.new(FIELD_W * 0.790, FIELD_H * 0.500)

	else
		playerPos = Vector2.new(FIELD_W * 0.210, FIELD_H * 0.500)
		botPos = Vector2.new(FIELD_W * 0.575, FIELD_H * 0.500)
		allyPos = Vector2.new(FIELD_W * 0.245, FIELD_H * 0.692)
		ally2Pos = Vector2.new(FIELD_W * 0.165, FIELD_H * 0.333)
		enemy2Pos = Vector2.new(FIELD_W * 0.755, FIELD_H * 0.317)
		enemy3Pos = Vector2.new(FIELD_W * 0.815, FIELD_H * 0.675)
	end

		actor.Pos = pos
		actor.Vel = Vector2.new(0, 0)
		actor.Facing = facing
		actor.Target = pos
		actor.Mode = "Support"

		actor.TargetIsBall = false
		actor.StuckTimer = 0
		actor.StuckScore = 0
		actor.LastPos = pos
		actor.LastBallPos = ballPos
		actor.Diff = diff
		actor.Active = active == true
	end

	resetExtra(ally2Actor, ally2Pos, 0, currentAllyDiff, teamSize >= 3)
	resetExtra(enemy2Actor, enemy2Pos, math.pi, currentDiff, teamSize >= 2)
	resetExtra(enemy3Actor, enemy3Pos, math.pi, currentDiff, teamSize >= 3)
	if UI.allyChip then UI.allyChip.Visible = allyActor.Active end
	if UI.ally2Chip then UI.ally2Chip.Visible = ally2Actor.Active end
	if UI.enemy2Chip then UI.enemy2Chip.Visible = enemy2Actor.Active end
	if UI.enemy3Chip then UI.enemy3Chip.Visible = enemy3Actor.Active end

	if cameraMode == "Close" then
		local focusPos = Net.GetLocalState()
		cameraCenter = focusPos * 0.78 + ballPos * 0.22
	else
		cameraCenter = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	end
	BotAI.botTargetPos = botPos
	BotAI.botMode = (kickoffTeam == "Bot") and "Attack" or "Challenge"
	BotAI.botCommitTimer = (kickoffTeam == "Bot") and 2.5 or 0
	BotAI.botPressTimer = 4
	BotAI.botRecoveryTimer = 0
	BotAI.botCornerTimer = 0
	BotAI.botTargetIsBall = false
	BotAI.ballWallTimer = 0

function finishMatch()
	state = "Finished"
	cancelCharge()
	UI.countdownLabel.Visible = false
	UI.goalBanner.Visible = false
	if UI.replayPanel then UI.replayPanel.Visible = false end
	if UI.replaySkipButton then UI.replaySkipButton.Visible = false end

	local playerWon
	if Net.InMatch then
		playerWon = (Net.WinnerTeam == Net.LocalTeam)
			or (Net.WinnerTeam == nil and ((Net.LocalTeam == "Blue" and playerScore > botScore) or (Net.LocalTeam == "Red" and botScore > playerScore)))
	else
		playerWon = playerScore > botScore
	end
	if playerWon then
		UI.finishTitle.Text = "🏆 ПОБЕДА!"
		UI.finishTitle.TextColor3 = COL.Yellow
	else
		UI.finishTitle.Text = "😤 ПОРАЖЕНИЕ"
		UI.finishTitle.TextColor3 = Color3.fromRGB(255, 110, 100)
	end
	UI.finishScore.Text = tostring(playerScore) .. "  :  " .. tostring(botScore)
	local reason
	local winGoalLimit = Net.InMatch and Net.WinGoals or SETTINGS.WIN_GOALS
	if Net.InMatch and Net.WinnerReason then
		reason = Net.WinnerReason
	elseif playerScore >= winGoalLimit or botScore >= winGoalLimit then
		reason = "Матч до " .. tostring(winGoalLimit) .. " голов завершён досрочно."
	elseif overtime then
		reason = "Всё решил ЗОЛОТОЙ ГОЛ в дополнительное время!"
	else
		reason = "Время матча вышло."
	end
	local shotText = (Net.InMatch or isTeamMode)
		and ("\nУдары: СИНИЕ " .. tostring(shotsPlayer) .. " — " .. tostring(shotsBot) .. " КРАСНЫЕ")
		or ("\nУдары: ВЫ " .. tostring(shotsPlayer) .. " — " .. tostring(shotsBot) .. " БОТ")

			total = #Net.Room.Players
			for _, info in ipairs(Net.Room.Players) do if info.Rematch then votes = votes + 1 end end
		end

			.. " • Реванш: " .. tostring(votes) .. "/" .. tostring(math.max(total, 1))

	elseif isTeamMode then
		diffText = "\nСоперники: " .. currentDiff.Title .. " • Напарник: " .. currentAllyDiff.Title
	else
		diffText = "\nСложность: " .. currentDiff.Title
	end

	UI.finishOverlay.Visible = true
end

-- Запись последних секунд матча для замедленного повтора.
local function recordReplayFrame(dt)
	if state ~= "Playing" and state ~= "Goal" then
		return
	end
	Replay.replayClock = Replay.replayClock + dt
	Replay.replaySampleTimer = Replay.replaySampleTimer - dt
	if Replay.replaySampleTimer > 0 then
		return
	end
	Replay.replaySampleTimer = 1 / 30
	table.insert(Replay.replayFrames, {
		t = Replay.replayClock,
		p = playerPos,
		b = botPos,
		ball = ballPos,
		rot = ballRotation,
		pf = playerFacingAngle,
		bf = botFacingAngle,
		a1 = allyActor.Pos,
		a1f = allyActor.Facing,
		a2 = ally2Actor.Pos,
		a2f = ally2Actor.Facing,
		e2 = enemy2Actor.Pos,
		e2f = enemy2Actor.Facing,
		e3 = enemy3Actor.Pos,
		e3f = enemy3Actor.Facing,
	})
	local cutoff = Replay.replayClock - (SETTINGS.REPLAY_SOURCE + 1.6)
	while #Replay.replayFrames > 2 and Replay.replayFrames[1].t < cutoff do
		table.remove(Replay.replayFrames, 1)
	end
end

local function startGoalReplay()
	if #Replay.replayFrames < 2 then
		completeGoalSequence()
		return
	end
	Replay.replayLiveState = {
		p = playerPos, b = botPos, ball = ballPos,
		pv = playerVel, bv = botVel, ballv = ballVel,
		rot = ballRotation, pf = playerFacingAngle, bf = botFacingAngle,
		a1 = allyActor.Pos, a1v = allyActor.Vel, a1f = allyActor.Facing,
		a2 = ally2Actor.Pos, a2v = ally2Actor.Vel, a2f = ally2Actor.Facing,
		e2 = enemy2Actor.Pos, e2v = enemy2Actor.Vel, e2f = enemy2Actor.Facing,
		e3 = enemy3Actor.Pos, e3v = enemy3Actor.Vel, e3f = enemy3Actor.Facing,
	}
	Replay.replaySourceEnd = Replay.replayFrames[#Replay.replayFrames].t
	Replay.replaySourceStart = math.max(Replay.replayFrames[1].t, Replay.replaySourceEnd - SETTINGS.REPLAY_SOURCE)
	Replay.replayFrameIndex = 1
	while Replay.replayFrameIndex < #Replay.replayFrames and Replay.replayFrames[Replay.replayFrameIndex + 1].t <= Replay.replaySourceStart do
		Replay.replayFrameIndex = Replay.replayFrameIndex + 1
	end
	local sourceDuration = Replay.replaySourceEnd - Replay.replaySourceStart
	if sourceDuration < 0.2 then
		completeGoalSequence()
		return
	end
	Replay.replayDuration = SETTINGS.REPLAY_DURATION -- ровно 7 секунд на экране
	Replay.replayTime = 0
	Replay.replayActive = true
	state = "Replay"
	UI.goalBanner.Visible = false
	UI.replayPanel.Visible = true
	UI.replaySkipButton.Visible = true
	UI.replayTitle.Text = replayTitleText()
	UI.replayProgressFill.Size = UDim2.fromScale(0, 1)
end

-- ЧИСТАЯ ФИЗИКА КОНТАКТА (БЕЗ КУРСОРА):
-- Направление удара = линия ЦЕНТР ФИШКИ -> ЦЕНТР МЯЧА в момент касания.
-- Подошёл слева — мяч откинется вправо. Подошёл справа — влево.
-- Подошёл снизу — вверх, сверху — вниз, по диагонали — по той же диагонали.
-- Курсор, старая скорость мяча и боковая скорость фишки направление НЕ меняют.
local function getContactDirection(isBot)
	local chipPos, chipVel
	if isBot then
		chipPos, chipVel = botPos, botVel
	elseif Net.InMatch then
		chipPos, chipVel = Net.GetLocalState()
	else
		chipPos, chipVel = playerPos, playerVel
	end
	local dir = unit2(ballPos - chipPos)
	if dir.Magnitude < 0.01 then
		dir = unit2(chipVel)
	end
	if dir.Magnitude < 0.01 then
		dir = isBot and Vector2.new(-1, 0) or Vector2.new(1, 0)
	end
	return dir
end

-- Медленная нелинейная сила ИГРОКА: начало шкалы очень мягкое.
-- Короткий клик даёт короткий пас, а настоящий выстрел надо честно держать.
local function getEffectivePlayerPower(rawPower)
	return math.pow(clamp(rawPower, 0, 1), 1.28)
end

-- Реалистичный ответ мяча на удар:
-- главная скорость идёт строго по линии контакта фишка -> мяч, а собственная
-- боковая скорость уже летящего мяча частично сохраняется. Стоячий мяч летит ровно по контакту.
local function applyRealisticKick(dir, targetNormalSpeed)
	local oldNormalSpeed = ballVel:Dot(dir)
	local tangentVel = ballVel - dir * oldNormalSpeed
	local outgoingNormal = math.max(targetNormalSpeed, oldNormalSpeed * 0.38)
	ballVel = dir * outgoingNormal + tangentVel * 0.38
	if ballVel.Magnitude > 1120 * FIELD_SCALE then
		ballVel = unit2(ballVel) * (1120 * FIELD_SCALE)
	end
end

local function pulseBall()
	task.spawn(function()
		UI.ballFrame.Size = fieldSizeToUI(BALL_R * 2 * 1.12, BALL_R * 2 * 1.12)
		task.wait(0.05)
		UI.ballFrame.Size = fieldSizeToUI(BALL_R * 2, BALL_R * 2)
	end)
end

-- Тот же физический удар для дополнительных ботов в командных режимах.
local function kickBallForActor(actor, power01)
	if not actor or not actor.Active then
		return false
	end
	local toBall = ballPos - actor.Pos
	local reach = PLAYER_R + BALL_R + BOT_KICK_REACH
	if toBall.Magnitude > reach then
		return false
	end
	local dir = unit2(toBall)
	if dir.Magnitude < 0.01 then
		dir = unit2(actor.Vel)
	end
	if dir.Magnitude < 0.01 then
		dir = (actor.Team == "Blue") and Vector2.new(1, 0) or Vector2.new(-1, 0)
	end
	ballPos = actor.Pos + dir * (PLAYER_R + BALL_R)
	local p = clamp(power01, 0, 1)
	local d = actor.Diff or DIFFICULTIES.Medium
	local speed = (230 + p * d.KickPower) * FIELD_SCALE + math.max(0, actor.Vel:Dot(dir)) * 0.30
	applyRealisticKick(dir, math.min(speed, 1120 * FIELD_SCALE))
	if actor.Team == "Blue" then
		shotsPlayer = shotsPlayer + 1
	else
		shotsBot = shotsBot + 1
	end
	playSfx(SFX.Kick, 0.18 + p * 0.18, 0.95 + p * 0.10)
	pulseBall()
	return true
end

-- Удар по мячу только при касании кругов. Возвращает true, если удар произошёл.
local function kickBall(isBot, power01)
	local chipPos, chipVel
	if isBot then
		chipPos, chipVel = botPos, botVel
	elseif Net.InMatch then
		chipPos, chipVel = Net.GetLocalState()
	else
		chipPos, chipVel = playerPos, playerVel
	end
	local toBall = ballPos - chipPos
	local reach = PLAYER_R + BALL_R + (isBot and BOT_KICK_REACH or PLAYER_KICK_REACH)
	if toBall.Magnitude > reach then
		return false
	end

	local dir = getContactDirection(isBot)
	-- Ставим мяч точно на поверхность контакта: никаких случайных перекрытий кругов.
	ballPos = chipPos + dir * (PLAYER_R + BALL_R)

	local p = clamp(power01, 0, 1)
	local speed
	local approachFactor
	if isBot then
		speed = (240 + p * currentDiff.KickPower) * FIELD_SCALE
		approachFactor = 0.35
	else
		local effective = getEffectivePlayerPower(p)
		speed = (150 + effective * 850) * FIELD_SCALE
		approachFactor = 0.08 + effective * 0.22
	end
	-- Только скорость фишки ВДОЛЬ линии контакта усиливает удар; боковая скорость его не уводит.
	local approachSpeed = math.max(0, chipVel:Dot(dir))
	speed = math.min(speed + approachSpeed * approachFactor, 1120 * FIELD_SCALE)
	applyRealisticKick(dir, speed)
	-- Никакого рывка/отдачи персонажа на ЛКМ нет: двигается только мяч.

	if isBot then
		shotsBot = shotsBot + 1
	else
		shotsPlayer = shotsPlayer + 1
	end
	playSfx(SFX.Kick, 0.18 + p * 0.18, 0.95 + p * 0.10)
	pulseBall()
	return true
end

-- Отпустили ЛКМ / кнопку УДАР
local function releaseKick()
	if not isCharging or state ~= "Playing" then
		cancelCharge()
		return
	end
	local power = math.max(chargePower, 0.06) -- совсем короткий клик = очень мягкий пас
	isCharging = false
	UI.powerBarBg.Visible = false
	UI.aimLine.Visible = false

	-- Онлайн удар считает сервер по настоящему контакту кругов.
	-- Оффлайн kickBall вернёт false без касания; рывка персонажа нет нигде.
	if Net.InMatch then
		if Net.SendKick then Net.SendKick(power) end
	else
		kickBall(false, power)
	end
	chargePower = 0
	chargeAimToGoal = false
end

--------------------------------------------------------------------------------
--  ИИ БОТА
--------------------------------------------------------------------------------
local function botStep(dt)
	BotAI.botThinkTimer = BotAI.botThinkTimer - dt
	BotAI.botKickCooldown = math.max(0, BotAI.botKickCooldown - dt)
	BotAI.botCommitTimer = math.max(0, BotAI.botCommitTimer - dt)
	BotAI.botPressTimer = BotAI.botPressTimer - dt
	if BotAI.botPressTimer <= 0 then
		-- Даже если игрок долго держит мяч на своей половине,
		-- бот периодически сам идёт в прессинг, а не ждёт у своих ворот.
		BotAI.botCommitTimer = math.max(BotAI.botCommitTimer, 2.3)
		BotAI.botPressTimer = rng:NextNumber(6.5, 9.5)
	end
	local d = currentDiff
	BotAI.botRecoveryTimer = math.max(0, BotAI.botRecoveryTimer - dt)

	local function startBotRecovery(fromPos, duration)
		if BotAI.botRecoveryTimer > 0 then
			return
		end
		BotAI.botRecoveryTimer = duration or 1.15
		local toCenter = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - fromPos)
		if toCenter.Magnitude < 0.01 then
			toCenter = Vector2.new(1, 0)
		end
		local _, wallInward = getNearestWall(fromPos)
		BotAI.botRecoveryPoint = clampBotTarget(fromPos + wallInward * (145 * FIELD_SCALE) + toCenter * (50 * FIELD_SCALE))
		BotAI.botTargetPos = BotAI.botRecoveryPoint
		BotAI.botTargetIsBall = false
		BotAI.botCornerTimer = 0
		BotAI.botStuckScore = 0
		BotAI.botThinkTimer = 0
	end

	-- Если мяч почти стоит у ЛЮБОГО борта, а бот его подпирает, бот обязан отойти.
	local stuckBallDist = (ballPos - botPos).Magnitude
	local ballAtWall = getWallBallInfo(ballPos)
	if ballAtWall and ballVel.Magnitude < 70 * FIELD_SCALE and stuckBallDist < 125 then
		BotAI.botCornerTimer = BotAI.botCornerTimer + dt
	else
		BotAI.botCornerTimer = math.max(0, BotAI.botCornerTimer - dt * 2)
	end
	local pursuingWallBall = ballAtWall and (BotAI.botTargetIsBall
		or BotAI.botMode == "Attack" or BotAI.botMode == "Challenge")
	if BotAI.botCornerTimer > 0.95 and not pursuingWallBall then
		-- Отойти должен только бот, который НЕ назначен выносить этот мяч.
		startBotRecovery(ballPos, 0.65)
	end

	-- Общий контроль застревания: бот хочет ехать, но за 0.45 сек почти не сдвинулся.
	BotAI.botStuckSampleTimer = BotAI.botStuckSampleTimer + dt
	if BotAI.botStuckSampleTimer >= 0.45 then
		local botMoved = (botPos - BotAI.botLastSamplePos).Magnitude
		local ballMoved = (ballPos - BotAI.botLastBallSamplePos).Magnitude
		local wantsToMove = (BotAI.botTargetPos - botPos).Magnitude > 18
		local wallBallSlow = ballAtWall and not pursuingWallBall and ballMoved < 11 * FIELD_SCALE and stuckBallDist < 145
		local botNearWall = getNearestWall(botPos) <= PLAYER_R + 10
		local pressedToWall = botNearWall and wantsToMove and botMoved < 14 * FIELD_SCALE
		if BotAI.botRecoveryTimer <= 0 and not pursuingWallBall
			and ((wantsToMove and botMoved < 12 * FIELD_SCALE) or wallBallSlow or pressedToWall) then
			BotAI.botStuckScore = BotAI.botStuckScore + 1
		else
			BotAI.botStuckScore = math.max(0, BotAI.botStuckScore - 1)
		end
		if BotAI.botStuckScore >= 2 then
			startBotRecovery(wallBallSlow and ballPos or botPos, wallBallSlow and 0.65 or 1.0)
		end
		BotAI.botLastSamplePos = botPos
		BotAI.botLastBallSamplePos = ballPos
		BotAI.botStuckSampleTimer = 0
	end

	-- «Мозг» бота просыпается раз в Reaction секунд (у сложного — чаще)
	if BotAI.botThinkTimer <= 0 then
		BotAI.botThinkTimer = d.Reaction

		-- Предсказание перехвата: где мяч окажется, пока бот до него добежит
		local distToBall = (ballPos - botPos).Magnitude
		local reachTime = clamp(distToBall / math.max(d.Speed * FIELD_SCALE, 1), 0, 0.85)
		local predicted = ballPos + ballVel * reachTime * ((0.25 + d.Predict) * 0.78)
		predicted = clampToField(predicted, BALL_R)

		local botDist = (predicted - botPos).Magnitude
		local playerDist = (predicted - playerPos).Magnitude
		local enemy2Dist = math.huge -- ближайший другой красный бот (в 2×2 это Бот-2, в 3×3 — Бот-2 или Бот-3)
		local nearestRedExtraPos = nil
		local allyDist = math.huge
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				local actorDistToPredicted = (predicted - actor.Pos).Magnitude
				if actor.Team == "Red" and actorDistToPredicted < enemy2Dist then
					enemy2Dist = actorDistToPredicted
					nearestRedExtraPos = actor.Pos
				elseif actor.Team == "Blue" and actorDistToPredicted < allyDist then
					allyDist = actorDistToPredicted
				end
			end
		end
		local botETA = botDist / math.max(d.Speed * FIELD_SCALE, 1)
		local playerETA = playerDist / (SETTINGS.PLAYER_SPRINT * FIELD_SCALE)
		local canWinRace = botETA <= playerETA + (-0.12 + d.Aggression * 0.55)

		local playerHasBall = (ballPos - playerPos).Magnitude <= PLAYER_R + BALL_R + 50
		local botHasBall = (ballPos - botPos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH + 28
		local allyHasBall = false
		local enemy2HasBall = false
		for _, actor in ipairs(extraActors) do
			if actor.Active and (ballPos - actor.Pos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH + 28 then
				if actor.Team == "Blue" then
					allyHasBall = true
				else
					enemy2HasBall = true
				end
			end
		end
		local blueHasBall = playerHasBall or allyHasBall
		local looseBall = not blueHasBall and not botHasBall and not enemy2HasBall
		local ballThreat = (ballPos.X > 720 * FIELD_SCALE and ballVel.X > 110 * FIELD_SCALE)
			or (ballPos.X > 850 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 175 * FIELD_SCALE)

		-- Линия прессинга: сложный бот встречает игрока очень глубоко.
		-- Проигрывает или мало времени — прессингует ещё выше.
		local pressLine = (470 - d.Aggression * 320) * FIELD_SCALE
		if botScore < playerScore then
			pressLine = pressLine - 110 * FIELD_SCALE
		end
		if overtime or (timeLeft < 35 and botScore <= playerScore) then
			pressLine = pressLine - 80 * FIELD_SCALE
		end

		if ballThreat then
			BotAI.botMode = "Defend"
		elseif botHasBall or BotAI.botCommitTimer > 0 or canWinRace
			or predicted.X >= pressLine
			or (looseBall and predicted.X >= pressLine - 110 * FIELD_SCALE)
			or BotAI.ballIdleTimer > 1.2 then
			BotAI.botMode = "Attack"
		elseif playerHasBall and predicted.X >= 260 * FIELD_SCALE then
			BotAI.botMode = "Challenge"
		else
			BotAI.botMode = "Defend"
		end

		-- Только ближайший красный идёт до конца в мяч. Если Бот-2 заметно ближе,
		-- главный бот открывается или страхует, а не создаёт давку у борта.
		local wallBallNow = getWallBallInfo(predicted)
		if wallBallNow and (botHasBall or botDist <= enemy2Dist + 5) then
			BotAI.botMode = "Attack" -- назначенный красный доводит мяч у стены до выноса
		end
		if nearestRedExtraPos and not botHasBall and not ballThreat and enemy2Dist + 5 < botDist then
			BotAI.botMode = blueHasBall and "Defend" or "Support"
		end
		BotAI.botTargetIsBall = false

		-- Куда бот ХОЧЕТ отправить мяч. Отсюда считается и заход за мяч:
		-- для физического удара бот встаёт С ПРОТИВОПОЛОЖНОЙ стороны.
		local attackTarget
		if isInCornerZone(predicted, 140) then
			-- Сначала вытащить мяч из угла в открытую зону, атака уже потом
			attackTarget = Vector2.new((predicted.X < FIELD_W / 2) and 300 * FIELD_SCALE or 700 * FIELD_SCALE, (predicted.Y < FIELD_H / 2) and 215 * FIELD_SCALE or 385 * FIELD_SCALE)
		elseif ballPos.X > 820 * FIELD_SCALE then
			attackTarget = Vector2.new(330 * FIELD_SCALE, (playerPos.Y < GOAL_CENTER_Y) and 490 * FIELD_SCALE or 110 * FIELD_SCALE)
		elseif predicted.X < d.ShootX * FIELD_SCALE then
			local cornerY = (playerPos.Y < GOAL_CENTER_Y) and (GOAL_Y2 - 18) or (GOAL_Y1 + 18)
			attackTarget = Vector2.new(0, cornerY)
		else
			local laneY = predicted.Y * 0.70 + GOAL_CENTER_Y * 0.30 + ((playerPos.Y < predicted.Y) and 38 or -38)
			attackTarget = Vector2.new(predicted.X - 320 * FIELD_SCALE, clamp(laneY, 70 * FIELD_SCALE, FIELD_H - 70 * FIELD_SCALE))
		end
		local shotDir = unit2(attackTarget - predicted)
		local approachPoint = predicted - shotDir * (PLAYER_R + BALL_R + 8)
		local approachReachable, safeApproachPoint = getReachableApproach(predicted, shotDir)
		local wallBall, wallApproachPoint = getWallExtraction(predicted, -1, attackTarget, botPos)
		if wallBall and not approachReachable then
			approachPoint = wallApproachPoint
		else
			approachPoint = safeApproachPoint
		end
		local contactDir = unit2(predicted - botPos)
		local alignment = contactDir:Dot(shotDir)

		if BotAI.botMode == "Attack" or BotAI.botMode == "Challenge" then
			-- Сначала заходим за мяч с правильной стороны, и только потом идём в мяч.
			-- Поэтому бот не толкает его случайно к своим воротам и реально ведёт атаку влево.
			if botDist > 92 or alignment < 0.20 then
				BotAI.botTargetPos = approachPoint
				BotAI.botTargetIsBall = false
			else
				BotAI.botTargetPos = predicted
				BotAI.botTargetIsBall = true
			end
		else
			if ballThreat then
				-- Угроза своим воротам: перехватываем мяч, по возможности со стороны ворот.
				if botDist > 80 and alignment < 0.05 then
					BotAI.botTargetPos = approachPoint
				BotAI.botTargetIsBall = false
				else
					BotAI.botTargetPos = predicted
				BotAI.botTargetIsBall = true
				end
			else
				-- Редкая настоящая оборона — и та высокой линией, а не стоя в воротах.
				local goalPos = Vector2.new(FIELD_W, GOAL_CENTER_Y)
				local dirToBall = unit2(ballPos - goalPos)
				local defPos = goalPos + dirToBall * (315 * FIELD_SCALE)
				BotAI.botTargetPos = Vector2.new(clamp(defPos.X, 540 * FIELD_SCALE, 925 * FIELD_SCALE), clamp(defPos.Y, 85 * FIELD_SCALE, FIELD_H - 85 * FIELD_SCALE))
				BotAI.botTargetIsBall = false
			end
		end

		if wallBall and not approachReachable and BotAI.botTargetPos == approachPoint then
			BotAI.botTargetIsBall = true -- финальная точка бокового касания у стены, не обычная навигационная цель
		end

		if BotAI.botMode == "Support" then
			local supportX = clamp(predicted.X - 215 * FIELD_SCALE, 150 * FIELD_SCALE, 735 * FIELD_SCALE)
			local openLaneY = (predicted.Y < GOAL_CENTER_Y) and 425 * FIELD_SCALE or 175 * FIELD_SCALE
			local supportY = clamp(predicted.Y * 0.35 + openLaneY * 0.65, 75 * FIELD_SCALE, FIELD_H - 75 * FIELD_SCALE)
			BotAI.botTargetPos = Vector2.new(supportX, supportY)
			if nearestRedExtraPos then
				BotAI.botTargetPos = separateBotTarget(BotAI.botTargetPos, nearestRedExtraPos, 104)
			end
			BotAI.botTargetPos = separateBotTarget(BotAI.botTargetPos, ballPos, 108)
			BotAI.botTargetIsBall = false
		end
	end

	if BotAI.botRecoveryTimer > 0 then
		BotAI.botMode = "Recover"
		BotAI.botTargetPos = BotAI.botRecoveryPoint
		BotAI.botTargetIsBall = false
	end
	BotAI.botTargetPos = BotAI.botTargetIsBall and clampToField(BotAI.botTargetPos, PLAYER_R) or clampBotTarget(BotAI.botTargetPos)

	-- Движение бота: тот же физический подход, что у игрока.
	local toTarget = BotAI.botTargetPos - botPos
	local targetDist = toTarget.Magnitude
	local speedNow = d.Speed * FIELD_SCALE
	if targetDist < 105 then
		speedNow = speedNow * clamp(targetDist / 105, 0.22, 1)
	end
	local ballDistNow = (ballPos - botPos).Magnitude
	if (BotAI.botMode == "Attack" or BotAI.botMode == "Challenge") and ballDistNow < 135 and not BotAI.botTargetIsBall then
		-- Рядом с мячом бот контролирует скорость, чтобы не пролетать мяч насквозь
		speedNow = math.min(speedNow, math.max(105 * FIELD_SCALE, d.Speed * FIELD_SCALE * 0.76))
	end
	local desired = Vector2.new(0, 0)
	if targetDist > 4 then
		desired = unit2(toTarget) * speedNow
	end
	desired = limitBotVelocityAtWall(botPos, desired, BotAI.botTargetIsBall)
	local botAccel = (d.Accel or 600) * FIELD_SCALE
	if desired.Magnitude < 1 then
		botAccel = botAccel * 1.15
	end
	botVel = approachVelocity(botVel, desired, botAccel, dt)
	if desired.Magnitude < 1 and botVel.Magnitude < 4 then
		botVel = Vector2.new(0, 0)
	end
	botPos = botPos + botVel * dt
	botPos, botVel = applyFieldClamp(botPos, botVel, PLAYER_R)

	-- Решение об ударе
	if BotAI.botRecoveryTimer <= 0 and BotAI.botKickCooldown <= 0 and (ballPos - botPos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH then
		local aim, power
		if isInCornerZone(ballPos, 140) then
			-- Не добиваем мяч в угол: цель только открытая зона внутри поля
			aim = Vector2.new((ballPos.X < FIELD_W / 2) and 300 * FIELD_SCALE or 700 * FIELD_SCALE, (ballPos.Y < FIELD_H / 2) and 215 * FIELD_SCALE or 385 * FIELD_SCALE)
			power = 0.62
		elseif ballPos.X > 820 * FIELD_SCALE then
			-- Опасно у своих ворот — вынос на фланг, подальше от игрока
			aim = Vector2.new(300 * FIELD_SCALE, (playerPos.Y < GOAL_CENTER_Y) and 490 * FIELD_SCALE or 110 * FIELD_SCALE)
			power = 0.76
		elseif ballPos.X < d.ShootX * FIELD_SCALE then
			-- Удар по воротам: в угол, дальний от игрока
			local cornerY = (playerPos.Y < GOAL_CENTER_Y) and (GOAL_Y2 - 18) or (GOAL_Y1 + 18)
			aim = Vector2.new(0, cornerY)
			power = 0.86
		else
			-- Атакующее ведение: толкаем мяч к воротам игрока и ближе к центру
			local laneY = ballPos.Y * 0.70 + GOAL_CENTER_Y * 0.30 + ((playerPos.Y < ballPos.Y) and 38 or -38)
			aim = Vector2.new(ballPos.X - 320 * FIELD_SCALE, clamp(laneY, 70 * FIELD_SCALE, FIELD_H - 70 * FIELD_SCALE))
			power = 0.50
		end

		local wallApproachPoint
		local wallBall, wallApproach, wallAim = getWallExtraction(ballPos, -1, aim, botPos)
		if wallBall then
			local normalReachable = getReachableApproach(ballPos, unit2(aim - ballPos))
			if not normalReachable then
				aim = wallAim
				power = 0.66
				wallApproachPoint = wallApproach
			end
		end

		-- Физический удар нельзя делать с неправильной стороны:
		-- если бот оказался не за мячом, он обежит его, а не ударит назад.
		local desiredDir = unit2(aim - ballPos)
		local contactDirNow = unit2(ballPos - botPos)
		local alignmentNow = contactDirNow:Dot(desiredDir)
		local emergency = ballPos.X > 845 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 180 * FIELD_SCALE
		local minAlignment = clamp(0.68 - d.AimError * 0.026, 0.02, 0.52) -- сложный бот ждёт более точный заход за мяч
		if isInCornerZone(ballPos, 140) then
			minAlignment = math.max(minAlignment, 0.45)
		end
		if wallApproachPoint then
			-- Для бокового выноса от стены важен компонент вдоль борта; стена вернёт мяч внутрь.
			-- Не ждём идеальную геометрию: мяч важнее вытащить в игру, чем красиво обходить его.
			minAlignment = math.min(minAlignment, (BotAI.botCornerTimer > 0.35) and 0.14 or 0.24)
		end

		if alignmentNow >= minAlignment or (emergency and alignmentNow > -0.35) then
			-- Бот тоже бьёт только по линии контакта своих кругов; неточность даёт сам заход, а не магия.
			if kickBall(true, power) then
				BotAI.botKickCooldown = d.KickCooldown * rng:NextNumber(0.85, 1.15)
				-- После ведения/выноса бот продолжает атаку за мячом
				BotAI.botCommitTimer = math.max(BotAI.botCommitTimer, (ballPos.X > 820 * FIELD_SCALE) and 1.7 or 2.3)
				BotAI.botMode = "Attack"
			end
		else
			if wallApproachPoint and (botPos - wallApproachPoint).Magnitude < 12 and BotAI.botCornerTimer > 0.35 then
				BotAI.botTargetPos = ballPos -- уже стоим у точки: доводим касанием, а не кружим вокруг
			else
				BotAI.botTargetPos = wallApproachPoint or clampBotTarget(ballPos - desiredDir * (PLAYER_R + BALL_R + 10))
			end
			BotAI.botTargetIsBall = wallApproachPoint ~= nil
			BotAI.botThinkTimer = math.min(BotAI.botThinkTimer, 0.05)
		end
	end
end

local function getActorTeammate(actor)
	-- Для паса выбираем самого выдвинутого партнёра, а не только капитана команды.
	local bestPos = (actor.Team == "Blue") and playerPos or botPos
	local bestVel = (actor.Team == "Blue") and playerVel or botVel
	local bestX = bestPos.X
	for _, other in ipairs(extraActors) do
		if other ~= actor and other.Active and other.Team == actor.Team then
			local better = (actor.Team == "Blue" and other.Pos.X > bestX)
				or (actor.Team == "Red" and other.Pos.X < bestX)
			if better then
				bestPos = other.Pos
				bestVel = other.Vel
				bestX = other.Pos.X
			end
		end
	end
	return bestPos, bestVel
end

local function getNearestOpponentPos(actor, fromPos)
	local bestPos = nil
	local bestDist = math.huge
	local function consider(pos)
		local dist = (pos - fromPos).Magnitude
		if dist < bestDist then
			bestDist = dist
			bestPos = pos
		end
	end
	if actor.Team == "Blue" then
		consider(botPos)
	else
		consider(playerPos)
	end
	for _, other in ipairs(extraActors) do
		if other.Active and other.Team ~= actor.Team then
			consider(other.Pos)
		end
	end
	return bestPos or fromPos, bestDist
end

local function chooseExtraActorAim(actor, fromPos)
	local d = actor.Diff or DIFFICULTIES.Medium
	local isBlue = actor.Team == "Blue"
	local attackDirX = isBlue and 1 or -1
	local attackGoalX = isBlue and FIELD_W or 0
	local teammatePos, teammateVel = getActorTeammate(actor)
	local nearestOpponent = getNearestOpponentPos(actor, fromPos)

	local wallBall, _, wallAim = getWallExtraction(fromPos, attackDirX, nil, actor.Pos)
	if wallBall then
		return wallAim, 0.66
	end

	if isInCornerZone(fromPos, 140) then
		return Vector2.new((fromPos.X < FIELD_W / 2) and 300 * FIELD_SCALE or 700 * FIELD_SCALE,
			(fromPos.Y < FIELD_H / 2) and 215 * FIELD_SCALE or 385 * FIELD_SCALE), 0.62
	end

	local dangerAtOwnGoal = (isBlue and fromPos.X < 190 * FIELD_SCALE) or ((not isBlue) and fromPos.X > 810 * FIELD_SCALE)
	if dangerAtOwnGoal then
		return Vector2.new(
			clamp(fromPos.X + attackDirX * (350 * FIELD_SCALE), 70 * FIELD_SCALE, FIELD_W - 70 * FIELD_SCALE),
			(nearestOpponent.Y < GOAL_CENTER_Y) and 485 * FIELD_SCALE or 115 * FIELD_SCALE
		), 0.76
	end

	local inShootZone = (isBlue and fromPos.X > FIELD_W - d.ShootX * FIELD_SCALE)
		or ((not isBlue) and fromPos.X < d.ShootX * FIELD_SCALE)
	if inShootZone then
		local cornerY = (nearestOpponent.Y < GOAL_CENTER_Y) and (GOAL_Y2 - 18) or (GOAL_Y1 + 18)
		return Vector2.new(attackGoalX, cornerY), 0.86
	end

	-- Пас свободному партнёру, если он впереди и линия паса не закрыта соперником.
	local teammateAhead = (isBlue and teammatePos.X > fromPos.X + 45)
		or ((not isBlue) and teammatePos.X < fromPos.X - 45)
	local passMiddle = (fromPos + teammatePos) * 0.5
	if teammateAhead and (passMiddle - nearestOpponent).Magnitude > 95 then
		local passTarget = teammatePos + teammateVel * 0.18 + Vector2.new(attackDirX * 28, 0)
		return clampToField(passTarget, BALL_R), 0.50
	end

	local laneY = fromPos.Y * 0.70 + GOAL_CENTER_Y * 0.30
		+ ((nearestOpponent.Y < fromPos.Y) and 36 or -36)
	return Vector2.new(
		clamp(fromPos.X + attackDirX * (315 * FIELD_SCALE), 60 * FIELD_SCALE, FIELD_W - 60 * FIELD_SCALE),
		clamp(laneY, 70 * FIELD_SCALE, FIELD_H - 70 * FIELD_SCALE)
	), 0.50
end

local function extraActorStep(actor, dt)
	if not actor.Active then
		return
	end
	local d = actor.Diff or DIFFICULTIES.Medium
	local isBlue = actor.Team == "Blue"
	local attackDirX = isBlue and 1 or -1
	actor.Think = actor.Think - dt
	actor.KickCooldown = math.max(0, actor.KickCooldown - dt)
	actor.Commit = math.max(0, actor.Commit - dt)
	actor.Recovery = math.max(0, actor.Recovery - dt)

	local function startRecovery(fromPos)
		if actor.Recovery > 0 then
			return
		end
		actor.Recovery = 0.75 -- коротко освобождаем проход и сразу возвращаемся в игру
		local toCenter = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - fromPos)
		if toCenter.Magnitude < 0.01 then
			toCenter = Vector2.new(attackDirX, 0)
		end
		local _, wallInward = getNearestWall(fromPos)
		actor.RecoveryPoint = clampBotTarget(fromPos + wallInward * (145 * FIELD_SCALE) + toCenter * (50 * FIELD_SCALE))
		actor.Target = actor.RecoveryPoint
		actor.TargetIsBall = false
		actor.CornerTimer = 0
		actor.StuckScore = 0
		actor.Think = 0
	end

	local distToBallNow = (ballPos - actor.Pos).Magnitude
	local ballAtWall = getWallBallInfo(ballPos)
	if ballAtWall and ballVel.Magnitude < 70 * FIELD_SCALE and distToBallNow < 125 then
		actor.CornerTimer = actor.CornerTimer + dt
	else
		actor.CornerTimer = math.max(0, actor.CornerTimer - dt * 2)
	end
	local pursuingWallBall = ballAtWall and (actor.TargetIsBall or actor.Mode == "Chase")
	if actor.CornerTimer > 0.95 and not pursuingWallBall then
		startRecovery(ballPos)
	end

	actor.StuckTimer = actor.StuckTimer + dt
	if actor.StuckTimer >= 0.45 then
		local moved = (actor.Pos - actor.LastPos).Magnitude
		local ballMoved = (ballPos - actor.LastBallPos).Magnitude
		local wantsToMove = (actor.Target - actor.Pos).Magnitude > 18
		local wallBallSlow = ballAtWall and not pursuingWallBall and ballMoved < 11 * FIELD_SCALE and distToBallNow < 145
		local actorNearWall = getNearestWall(actor.Pos) <= PLAYER_R + 10
		local pressedToWall = actorNearWall and wantsToMove and moved < 14 * FIELD_SCALE
		if actor.Recovery <= 0 and not pursuingWallBall
			and ((wantsToMove and moved < 12 * FIELD_SCALE) or wallBallSlow or pressedToWall) then
			actor.StuckScore = actor.StuckScore + 1
		else
			actor.StuckScore = math.max(0, actor.StuckScore - 1)
		end
		if actor.StuckScore >= 2 then
			startRecovery(wallBallSlow and ballPos or actor.Pos)
		end
		actor.LastPos = actor.Pos
		actor.LastBallPos = ballPos
		actor.StuckTimer = 0
	end

	if actor.Think <= 0 then
		actor.Think = d.Reaction
		local distToBall = (ballPos - actor.Pos).Magnitude
		local reachTime = clamp(distToBall / math.max(d.Speed * FIELD_SCALE, 1), 0, 0.85)
		local predicted = clampToField(
			ballPos + ballVel * reachTime * ((0.25 + d.Predict) * 0.78), BALL_R)
		local teammatePos = getActorTeammate(actor)
		local primaryTeammatePos = isBlue and playerPos or botPos
		local actorDist = (predicted - actor.Pos).Magnitude
		local _, opponentDist = getNearestOpponentPos(actor, ballPos)
		local primaryTeammateDist = (predicted - primaryTeammatePos).Magnitude
		local actorClosest
		if actor.Team == "Red" then
			actorClosest = actorDist + 5 < primaryTeammateDist -- у Бота-1 приоритет при почти равной дистанции
		else
			actorClosest = actorDist <= primaryTeammateDist + 20
		end
		local teammateHasBall = (ballPos - primaryTeammatePos).Magnitude <= PLAYER_R + BALL_R + 42
		for _, other in ipairs(extraActors) do
			if other ~= actor and other.Active and other.Team == actor.Team then
				local otherDist = (predicted - other.Pos).Magnitude
				if (ballPos - other.Pos).Magnitude <= PLAYER_R + BALL_R + 42 then
					teammateHasBall = true
				end
				if otherDist + 5 < actorDist
					or (math.abs(otherDist - actorDist) <= 5 and (other.Priority or 9) < (actor.Priority or 9)) then
					actorClosest = false
				end
			end
		end
		local actorHasBall = distToBall <= PLAYER_R + BALL_R + BOT_KICK_REACH + 18
		local opponentHasBall = opponentDist <= PLAYER_R + BALL_R + 42
		local looseBall = not actorHasBall and not teammateHasBall and not opponentHasBall
		local threatToOwnGoal
		if isBlue then
			threatToOwnGoal = (ballPos.X < 300 * FIELD_SCALE and ballVel.X < -110 * FIELD_SCALE)
				or (ballPos.X < 170 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 190 * FIELD_SCALE)
		else
			threatToOwnGoal = (ballPos.X > 700 * FIELD_SCALE and ballVel.X > 110 * FIELD_SCALE)
				or (ballPos.X > 830 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 190 * FIELD_SCALE)
		end

		local wallBallNow = getWallBallInfo(predicted)
		if actorHasBall or (wallBallNow and actorClosest) or (threatToOwnGoal and actorClosest)
			or (BotAI.ballIdleTimer > 1.2 and actorClosest)
			or (actorClosest and (looseBall or opponentHasBall or actor.Commit > 0)) then
			actor.Mode = "Chase"
		elseif teammateHasBall then
			actor.Mode = "Support"
		elseif threatToOwnGoal or opponentHasBall then
			actor.Mode = "Cover"
		else
			actor.Mode = "Support"
		end
		actor.TargetIsBall = false

		if actor.Mode == "Chase" then
			local wallBall, wallApproachPoint = getWallExtraction(predicted, attackDirX, nil, actor.Pos)
			if wallBall then
				actor.Target = wallApproachPoint
				actor.TargetIsBall = true -- финальная точка бокового касания у стены
			else
				local aim = chooseExtraActorAim(actor, predicted)
				local shotDir = unit2(aim - predicted)
				local _, approachPoint = getReachableApproach(predicted, shotDir)
				local alignment = unit2(predicted - actor.Pos):Dot(shotDir)
				if actorDist > 92 or alignment < 0.20 then
					actor.Target = approachPoint
					actor.TargetIsBall = false
				else
					actor.Target = predicted
					actor.TargetIsBall = true
				end
			end
		elseif actor.Mode == "Cover" then
			local ownGoal = Vector2.new(isBlue and 0 or FIELD_W, GOAL_CENTER_Y)
			local coverPos = ownGoal + unit2(ballPos - ownGoal) * (285 * FIELD_SCALE)
			if (actor.Priority or 1) >= 2 then
				coverPos = coverPos + Vector2.new(-attackDirX * (55 * FIELD_SCALE), (ballPos.Y < GOAL_CENTER_Y) and 58 * FIELD_SCALE or -58 * FIELD_SCALE)
			end
			if isBlue then
				actor.Target = Vector2.new(clamp(coverPos.X, 90 * FIELD_SCALE, 470 * FIELD_SCALE), clamp(coverPos.Y, 85 * FIELD_SCALE, FIELD_H - 85 * FIELD_SCALE))
			else
				actor.Target = Vector2.new(clamp(coverPos.X, 530 * FIELD_SCALE, 910 * FIELD_SCALE), clamp(coverPos.Y, 85 * FIELD_SCALE, FIELD_H - 85 * FIELD_SCALE))
			end
		else
			local supportX
			if teammateHasBall then
				supportX = isBlue and clamp(ballPos.X + 225 * FIELD_SCALE, 270 * FIELD_SCALE, 850 * FIELD_SCALE) or clamp(ballPos.X - 225 * FIELD_SCALE, 150 * FIELD_SCALE, 730 * FIELD_SCALE)
			else
				supportX = isBlue and clamp(predicted.X + 135 * FIELD_SCALE, 180 * FIELD_SCALE, 760 * FIELD_SCALE) or clamp(predicted.X - 135 * FIELD_SCALE, 240 * FIELD_SCALE, 820 * FIELD_SCALE)
			end
			local openLaneY = (ballPos.Y < GOAL_CENTER_Y) and 425 * FIELD_SCALE or 175 * FIELD_SCALE
			local supportY = clamp(predicted.Y * 0.35 + openLaneY * 0.65, 75 * FIELD_SCALE, FIELD_H - 75 * FIELD_SCALE)
			if (actor.Priority or 1) >= 2 then
				-- Второй напарник в 3×3 держит другую линию и чуть глубже страхует отскок.
				supportX = clamp(supportX - attackDirX * (105 * FIELD_SCALE), 90 * FIELD_SCALE, FIELD_W - 90 * FIELD_SCALE)
				local secondLaneY = (ballPos.Y < GOAL_CENTER_Y) and 175 * FIELD_SCALE or 425 * FIELD_SCALE
				supportY = clamp(predicted.Y * 0.50 + secondLaneY * 0.50, 75 * FIELD_SCALE, FIELD_H - 75 * FIELD_SCALE)
			end
			actor.Target = Vector2.new(supportX, supportY)
			actor.TargetIsBall = false
		end

		-- Не толпимся в одной точке с партнёрами, но и не прячемся далеко от мяча.
		if not actor.TargetIsBall then
			actor.Target = separateBotTarget(actor.Target, primaryTeammatePos, 104)
			for _, other in ipairs(extraActors) do
				if other ~= actor and other.Active and other.Team == actor.Team then
					actor.Target = separateBotTarget(actor.Target, other.Pos, 104)
				end
			end
		end
		if actor.Mode ~= "Chase" then
			actor.Target = separateBotTarget(actor.Target, ballPos, 108)
			actor.TargetIsBall = false
		end
	end

	if actor.Recovery > 0 then
		actor.Mode = "Recover"
		actor.Target = actor.RecoveryPoint
		actor.TargetIsBall = false
	end
	actor.Target = actor.TargetIsBall and clampToField(actor.Target, PLAYER_R) or clampBotTarget(actor.Target)

	local toTarget = actor.Target - actor.Pos
	local targetDist = toTarget.Magnitude
	local speedNow = d.Speed * FIELD_SCALE
	if targetDist < 105 then
		speedNow = speedNow * clamp(targetDist / 105, 0.22, 1)
	end
	if actor.Mode == "Chase" and distToBallNow < 135 and not actor.TargetIsBall then
		speedNow = math.min(speedNow, math.max(105 * FIELD_SCALE, d.Speed * FIELD_SCALE * 0.76))
	end
	local desired = Vector2.new(0, 0)
	if targetDist > 4 then
		desired = unit2(toTarget) * speedNow
	end
	desired = limitBotVelocityAtWall(actor.Pos, desired, actor.TargetIsBall)
	local accel = (d.Accel or 600) * FIELD_SCALE
	if desired.Magnitude < 1 then
		accel = accel * 1.15
	end
	actor.Vel = approachVelocity(actor.Vel, desired, accel, dt)
	if desired.Magnitude < 1 and actor.Vel.Magnitude < 4 then
		actor.Vel = Vector2.new(0, 0)
	end
	actor.Pos = actor.Pos + actor.Vel * dt
	actor.Pos, actor.Vel = applyFieldClamp(actor.Pos, actor.Vel, PLAYER_R)

	if actor.Recovery <= 0 and actor.KickCooldown <= 0
		and (ballPos - actor.Pos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH then
		local aim, power = chooseExtraActorAim(actor, ballPos)
		local wallApproachPoint
		local wallBall, wallApproach = getWallExtraction(ballPos, attackDirX, aim, actor.Pos)
		if wallBall then
			wallApproachPoint = wallApproach
		end
		local desiredDir = unit2(aim - ballPos)
		local contactDir = unit2(ballPos - actor.Pos)
		local alignment = contactDir:Dot(desiredDir)
		local emergency
		if isBlue then
			emergency = ballPos.X < 155 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 180 * FIELD_SCALE
		else
			emergency = ballPos.X > 845 * FIELD_SCALE and math.abs(ballPos.Y - GOAL_CENTER_Y) < 180 * FIELD_SCALE
		end
		local minAlignment = clamp(0.68 - d.AimError * 0.026, 0.02, 0.52)
		if isInCornerZone(ballPos, 140) then
			minAlignment = math.max(minAlignment, 0.45)
		end
		if wallApproachPoint then
			minAlignment = math.min(minAlignment, (actor.CornerTimer > 0.35) and 0.14 or 0.24)
		end
		if alignment >= minAlignment or (emergency and alignment > -0.35) then
			if kickBallForActor(actor, power) then
				actor.KickCooldown = d.KickCooldown * rng:NextNumber(0.85, 1.15)
				actor.Commit = math.max(actor.Commit, 1.8)
				actor.Mode = "Chase"
			end
		else
			if wallApproachPoint and (actor.Pos - wallApproachPoint).Magnitude < 12 and actor.CornerTimer > 0.35 then
				actor.Target = ballPos
			else
				actor.Target = wallApproachPoint or clampBotTarget(ballPos - desiredDir * (PLAYER_R + BALL_R + 10))
			end
			actor.TargetIsBall = wallApproachPoint ~= nil
			actor.Think = math.min(actor.Think, 0.05)
		end
	end
end

local function playerStep(dt)
	local move = Vector2.new(0, 0)
	local touchSpeedScale = 1
	if UI.joystickActive and UI.joystickVector and UI.joystickVector.Magnitude > 0.01 then
		-- Аналоговый стик: направление задаёт ручка, а величина отклонения — скорость.
		move = unit2(UI.joystickVector)
		touchSpeedScale = clamp(UI.joystickVector.Magnitude, 0, 1)
	elseif touchActive then
		local toTouch = touchFieldPos - playerPos
		if toTouch.Magnitude > 12 then
			move = unit2(toTouch)
			-- У пальца не дёргаемся: чем ближе цель, тем мягче скорость
			touchSpeedScale = clamp(toTouch.Magnitude / (95 * FIELD_SCALE), 0.25, 1)
		end
	elseif not UserInputService:GetFocusedTextBox() then
		if keysDown[Enum.KeyCode.W] or keysDown[Enum.KeyCode.Up] then move = move + Vector2.new(0, -1) end
		if keysDown[Enum.KeyCode.S] or keysDown[Enum.KeyCode.Down] then move = move + Vector2.new(0, 1) end
		if keysDown[Enum.KeyCode.A] or keysDown[Enum.KeyCode.Left] then move = move + Vector2.new(-1, 0) end
		if keysDown[Enum.KeyCode.D] or keysDown[Enum.KeyCode.Right] then move = move + Vector2.new(1, 0) end
	end

	local speed = SETTINGS.PLAYER_SPEED * FIELD_SCALE
	if keysDown[Enum.KeyCode.LeftShift] or keysDown[Enum.KeyCode.RightShift] then
		speed = SETTINGS.PLAYER_SPRINT * FIELD_SCALE
	end
	if isCharging then
		speed = speed * 0.72 -- пока держишь удар, идёшь спокойнее и точнее подходишь к мячу
	end
	speed = speed * touchSpeedScale

	local desired = Vector2.new(0, 0)
	if move.Magnitude > 0.01 then
		desired = unit2(move) * speed
	end

	-- Разгон ~0.35 сек, остановка с коротким скольжением, разворот с инерцией
	local acceleration = 820 * FIELD_SCALE
	if desired.Magnitude < 1 then
		acceleration = 980 * FIELD_SCALE
	elseif playerVel.Magnitude > 20 * FIELD_SCALE and playerVel:Dot(desired) < 0 then
		acceleration = 900 * FIELD_SCALE
	end
	playerVel = approachVelocity(playerVel, desired, acceleration, dt)
	if desired.Magnitude < 1 and playerVel.Magnitude < 5 then
		playerVel = Vector2.new(0, 0)
	end

	playerPos = playerPos + playerVel * dt
	playerPos, playerVel = applyFieldClamp(playerPos, playerVel, PLAYER_R)
end

-- Фишки не проходят друг сквозь друга (проверяются все пары активных футболистов)
local function resolvePair(aPos, aVel, bPos, bVel)
	local diff = bPos - aPos
	local dist = diff.Magnitude
	local minDist = PLAYER_R * 2
	if dist >= minDist then
		return aPos, aVel, bPos, bVel
	end
	local n
	if dist <= 0.001 then
		n = unit2(bVel - aVel)
		if n.Magnitude < 0.01 then
			n = Vector2.new(1, 0)
		end
	else
		n = diff / dist
	end
	local push = n * ((minDist - dist) / 2)
	aPos = aPos - push
	bPos = bPos + push
	local closingSpeed = (bVel - aVel):Dot(n)
	if closingSpeed < 0 then
		aVel = aVel + n * (closingSpeed / 2)
		bVel = bVel - n * (closingSpeed / 2)
	end
	return clampToField(aPos, PLAYER_R), aVel, clampToField(bPos, PLAYER_R), bVel
end

local function resolveChipChip()
	for _ = 1, 4 do
		playerPos, playerVel, botPos, botVel = resolvePair(playerPos, playerVel, botPos, botVel)
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				playerPos, playerVel, actor.Pos, actor.Vel =
					resolvePair(playerPos, playerVel, actor.Pos, actor.Vel)
				botPos, botVel, actor.Pos, actor.Vel =
					resolvePair(botPos, botVel, actor.Pos, actor.Vel)
			end
		end
		for i = 1, #extraActors - 1 do
			local actor = extraActors[i]
			if actor.Active then
				for j = i + 1, #extraActors do
					local other = extraActors[j]
					if other.Active then
						actor.Pos, actor.Vel, other.Pos, other.Vel =
							resolvePair(actor.Pos, actor.Vel, other.Pos, other.Vel)
					end
				end
			end
		end
	end
end

local function inGoalMouth(y)
	return y > GOAL_Y1 and y < GOAL_Y2
end

-- Во время позиционного решателя мяч нельзя случайно увести за борт.
-- В проёме ворот X не зажимаем: гол должен честно засчитываться линией ниже.
local function keepBallInsideForSolver()
	if inGoalMouth(ballPos.Y) then
		ballPos = Vector2.new(ballPos.X, clamp(ballPos.Y, BALL_R, FIELD_H - BALL_R))
	else
		ballPos = clampToField(ballPos, BALL_R)
	end
end

-- Минимальное расстояние от точки до центра любой активной фишки.
local function getBallChipClearance(pos)
	local best = math.min((pos - playerPos).Magnitude, (pos - botPos).Magnitude)
	for _, actor in ipairs(extraActors) do
		if actor.Active then
			best = math.min(best, (pos - actor.Pos).Magnitude)
		end
	end
	return best
end

-- Аварийный поиск ближайшей точки, где мяч гарантированно не пересекается
-- ни с одной фишкой. Используется только если обычное выталкивание уперлось
-- в стену/другие фишки или медленный мяч зажат давкой из нескольких кругов.
local function findFreeBallPosition(origin, preferredDir)
	local bestPos = nil
	local bestScore = -math.huge
	local minClearance = PLAYER_R + BALL_R + 1.25
	local function consider(rawPos)
		local candidate = clampToField(rawPos, BALL_R + 1)
		local clearance = getBallChipClearance(candidate)
		if clearance < minClearance then
			return
		end
		local wallClearance = math.min(candidate.X, FIELD_W - candidate.X, candidate.Y, FIELD_H - candidate.Y)
		local offset = candidate - origin
		local travel = offset.Magnitude
		local directionScore = 0
		if preferredDir and preferredDir.Magnitude > 0.01 and travel > 0.01 then
			directionScore = (offset / travel):Dot(unit2(preferredDir)) * 34
		end
		-- Важнее ближайшая безопасная точка, а не далёкий «идеальный» телепорт:
		-- мяч должен выскочить из давки рядом с борьбой, а не прыгнуть через полполя.
		local score = math.min(clearance, 78) * 0.82
			+ math.min(wallClearance, 90) * 0.28
			- travel * 1.05
			+ directionScore * 0.55
		if score > bestScore then
			bestScore = score
			bestPos = candidate
		end
	end
	for _, radius in ipairs({ 46, 60, 76, 94, 118, 150, 190 }) do
		for i = 0, 23 do
			local angle = (i / 24) * math.pi * 2
			consider(origin + Vector2.new(math.cos(angle), math.sin(angle)) * radius)
		end
	end
	consider(origin:Lerp(Vector2.new(FIELD_W / 2, FIELD_H / 2), 0.32))
	return bestPos
end

-- Один контакт мяч↔фишка. ВНИМАНИЕ: возвращает обновлённые pos/vel фишки.
-- Старый вариант двигал только мяч точно на окружность одной фишки; следующая
-- фишка могла тут же вернуть его внутрь первой. Теперь коррекция делится:
-- мяч получает большую часть сдвига, а фишка тоже физически уступает место.
local function collideBallWithChip(chipPos, chipVel)
	local diff = ballPos - chipPos
	local dist = diff.Magnitude
	local minDist = PLAYER_R + BALL_R
	if dist >= minDist then
		return chipPos, chipVel, false
	end
	local n
	if dist > 0.001 then
		n = diff / dist
	else
		-- Центры совпали: направление берём из относительного движения,
		-- затем из движения фишки и только в конце — к центру поля.
		n = unit2(ballVel - chipVel)
		if n.Magnitude < 0.01 then
			n = unit2(chipVel)
		end
		if n.Magnitude < 0.01 then
			n = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - chipPos)
		end
		if n.Magnitude < 0.01 then
			n = Vector2.new(1, 0)
		end
	end
	local penetration = minDist - dist
	ballPos = ballPos + n * (penetration * 0.74)
	chipPos = chipPos - n * (penetration * 0.26)
	chipPos, chipVel = applyFieldClamp(chipPos, chipVel, PLAYER_R)
	keepBallInsideForSolver()
	local relativeVel = ballVel - chipVel
	local relativeNormal = relativeVel:Dot(n)
	if relativeNormal < 0 then
		local relativeTangent = relativeVel - n * relativeNormal
		local chipNormalSpeed = chipVel:Dot(n)
		local restitution = (chipNormalSpeed > 35 * FIELD_SCALE) and 0.16 or 0.38
		ballVel = chipVel + relativeTangent * 0.88 - n * relativeNormal * restitution
	end
	return chipPos, chipVel, true
end

-- ГЛАВНАЯ ГАРАНТИЯ: после этой функции мяч не должен заканчивать шаг внутри
-- ни одной фишки. Контакты решаются несколькими проходами сразу со всеми
-- игроками, а не одним последовательным «поставил на окружность и забыл».
local function resolveBallChipOverlaps()
	local minDist = PLAYER_R + BALL_R
	for _ = 1, 12 do
		local anyHit = false
		local hit
		playerPos, playerVel, hit = collideBallWithChip(playerPos, playerVel)
		anyHit = anyHit or hit
		botPos, botVel, hit = collideBallWithChip(botPos, botVel)
		anyHit = anyHit or hit
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				actor.Pos, actor.Vel, hit = collideBallWithChip(actor.Pos, actor.Vel)
				anyHit = anyHit or hit
			end
		end
		if not anyHit then
			return true
		end
		resolveChipChip()
		keepBallInsideForSolver()
		if getBallChipClearance(ballPos) >= minDist - 0.05 then
			return true
		end
	end
	if getBallChipClearance(ballPos) < minDist - 0.05 then
		local freePos = findFreeBallPosition(ballPos, Vector2.new(FIELD_W / 2, FIELD_H / 2) - ballPos)
		if freePos then
			ballPos = freePos
			keepBallInsideForSolver()
		end
	end
	return getBallChipClearance(ballPos) >= minDist - 0.05
end

-- Swept circle-vs-circle: мяч движется не «телепортом в конечную точку», а
-- отрезком. Находим самое раннее касание с фишкой на этом отрезке, поэтому
-- быстрый мяч не пролетает сквозь круг даже при низком FPS/большом поле.
local function moveBallSafely(sdt)
	local minDist = PLAYER_R + BALL_R
	local timeLeft = sdt
	for _ = 1, 4 do
		local chips = {
			{ p = playerPos, v = playerVel },
			{ p = botPos, v = botVel },
		}
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				table.insert(chips, { p = actor.Pos, v = actor.Vel })
			end
		end
		local move = ballVel * timeLeft
		if move.Magnitude < 0.0001 then
			return
		end
		local earliest = 2
		local hitChip = nil
		for _, chip in ipairs(chips) do
			local rel = ballPos - chip.p
			local a = move:Dot(move)
			if a > 0.000001 then
				local b = 2 * rel:Dot(move)
				local c = rel:Dot(rel) - minDist * minDist
				local t = nil
				if c <= 0 then
					local n0 = unit2(rel)
					if n0.Magnitude > 0.01 and (ballVel - chip.v):Dot(n0) < -0.01 then
						t = 0
					end
				else
					local discriminant = b * b - 4 * a * c
					if discriminant >= 0 then
						t = (-b - math.sqrt(discriminant)) / (2 * a)
					end
				end
				if t and t >= 0 and t <= 1 and t < earliest then
					earliest = t
					hitChip = chip
				end
			end
		end
		if not hitChip then
			ballPos = ballPos + move
			return
		end
		ballPos = ballPos + move * earliest
		local n = unit2(ballPos - hitChip.p)
		if n.Magnitude < 0.01 then
			n = unit2(ballVel - hitChip.v)
		end
		if n.Magnitude < 0.01 then
			n = Vector2.new(1, 0)
		end
		ballPos = hitChip.p + n * (minDist + 0.04)
		keepBallInsideForSolver()
		local relativeVel = ballVel - hitChip.v
		local relativeNormal = relativeVel:Dot(n)
		if relativeNormal < 0 then
			local relativeTangent = relativeVel - n * relativeNormal
			local chipNormalSpeed = hitChip.v:Dot(n)
			local restitution = (chipNormalSpeed > 35 * FIELD_SCALE) and 0.16 or 0.38
			ballVel = hitChip.v + relativeTangent * 0.88 - n * relativeNormal * restitution
		end
		resolveBallChipOverlaps()
		timeLeft = timeLeft * (1 - earliest)
		if timeLeft <= 0.00001 then
			return
		end
	end
end

-- Медленный мяч, зажатый минимум двумя фишками и почти не двигающийся,
-- физически выдавливается в свободную сторону. Это отдельный предохранитель
-- именно для скриншота с БОТ 1 + БОТ 2 + напарник в одной точке.
local function updateBallClusterEscape(dt)
	BotAI.ballClusterEscapeCooldown = math.max(0, BotAI.ballClusterEscapeCooldown - dt)
	local contactCount = 0
	local normalSum = Vector2.new(0, 0)
	local function considerChip(pos)
		local diff = ballPos - pos
		local dist = diff.Magnitude
		if dist <= PLAYER_R + BALL_R + 24 then
			contactCount = contactCount + 1
			if dist > 0.01 then
				normalSum = normalSum + diff / dist
			end
		end
	end
	considerChip(playerPos)
	considerChip(botPos)
	for _, actor in ipairs(extraActors) do
		if actor.Active then
			considerChip(actor.Pos)
		end
	end
	local crowded = contactCount >= 2 and ballVel.Magnitude < 180 * FIELD_SCALE
	if crowded then
		BotAI.ballClusterCheckTimer = BotAI.ballClusterCheckTimer + dt
		if BotAI.ballClusterCheckTimer >= 0.25 then
			local moved = (ballPos - BotAI.ballClusterLastPos).Magnitude
			if moved < 15 then
				BotAI.ballClusterTimer = BotAI.ballClusterTimer + BotAI.ballClusterCheckTimer
			else
				BotAI.ballClusterTimer = math.max(0, BotAI.ballClusterTimer - BotAI.ballClusterCheckTimer)
			end
			BotAI.ballClusterLastPos = ballPos
			BotAI.ballClusterCheckTimer = 0
		end
	else
		BotAI.ballClusterTimer = math.max(0, BotAI.ballClusterTimer - dt * 2)
		BotAI.ballClusterCheckTimer = 0
		BotAI.ballClusterLastPos = ballPos
	end
	local neededTime = (contactCount >= 3) and 0.48 or 0.72
	if crowded and BotAI.ballClusterEscapeCooldown <= 0 and BotAI.ballClusterTimer >= neededTime then
		local preferred = normalSum
		if preferred.Magnitude < 0.25 then
			preferred = Vector2.new(FIELD_W / 2, FIELD_H / 2) - ballPos
		end
		if ballVel.Magnitude > 25 * FIELD_SCALE then
			preferred = preferred + unit2(ballVel) * 0.45
		end
		local oldPos = ballPos
		local freePos = findFreeBallPosition(oldPos, preferred)
		if freePos then
			local escapeDir = unit2(freePos - oldPos)
			if escapeDir.Magnitude < 0.01 then
				escapeDir = unit2(preferred)
			end
			if escapeDir.Magnitude < 0.01 then
				escapeDir = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - oldPos)
			end
			ballPos = freePos
			ballVel = escapeDir * (520 * FIELD_SCALE)
			BotAI.ballClusterTimer = 0
			BotAI.ballClusterCheckTimer = 0
			BotAI.ballClusterLastPos = ballPos
			BotAI.ballClusterEscapeCooldown = 1.5
			BotAI.ballWallTimer = 0
			BotAI.ballIdleTimer = 0
			playSfx(SFX.Bounce, 0.05, 0.95)
		end
	end
end

local function bounceBallOffNormal(n, restitution, tangentKeep)
	local normalSpeed = ballVel:Dot(n)
	if normalSpeed >= 0 then
		return 0
	end
	local tangentVel = ballVel - n * normalSpeed
	ballVel = tangentVel * tangentKeep - n * normalSpeed * restitution
	return -normalSpeed
end

local function collideBallCorner(sx, sy)
	local dx = (sx > 0) and ballPos.X or (FIELD_W - ballPos.X)
	local dy = (sy > 0) and ballPos.Y or (FIELD_H - ballPos.Y)
	if dx > 190 * FIELD_SCALE or dy > 190 * FIELD_SCALE then
		return
	end
	local need = CORNER_CUT + BALL_R * math.sqrt(2)
	local sum = dx + dy
	if sum >= need then
		return
	end
	local inward = Vector2.new(sx, sy) / math.sqrt(2)
	ballPos = ballPos + inward * ((need - sum) / math.sqrt(2))
	local impactSpeed = bounceBallOffNormal(inward, 0.64, 0.92)
	if impactSpeed > 90 * FIELD_SCALE then
		playSfx(SFX.Bounce, 0.03, 1.1)
	end
end

local function ballStep(dt)
	-- Сначала убираем перекрытия, созданные движением самих фишек в этом кадре,
	-- затем проверяем давку и только потом двигаем мяч swept-отрезками.
	resolveBallChipOverlaps()
	updateBallClusterEscape(dt)
	local steps = clamp(math.ceil((ballVel.Magnitude * dt) / 7), 4, 12)
	for _ = 1, steps do
		local sdt = dt / steps
		resolveBallChipOverlaps()

		-- Мёртвый мяч у борта: сначала лёгкое выкатывание, затем один заметный отскок.
		-- Смотрим не только скорость, но и реальное перемещение: катящийся вдоль стены мяч не трогаем.
		BotAI.ballWallEscapeCooldown = math.max(0, BotAI.ballWallEscapeCooldown - sdt)
		local ballAtWall, wallDist, wallInward = getWallBallInfo(ballPos)
		local touchingWall = ballAtWall and wallDist <= BALL_R + 9
		if touchingWall and ballVel.Magnitude < 150 * FIELD_SCALE then
			BotAI.ballWallCheckTimer = BotAI.ballWallCheckTimer + sdt
			if BotAI.ballWallCheckTimer >= 0.30 then
				local movedNearWall = (ballPos - BotAI.ballWallLastPos).Magnitude
				if movedNearWall < 16 * FIELD_SCALE then
					BotAI.ballWallTimer = BotAI.ballWallTimer + BotAI.ballWallCheckTimer
				else
					BotAI.ballWallTimer = math.max(0, BotAI.ballWallTimer - BotAI.ballWallCheckTimer * 1.5)
				end
				BotAI.ballWallLastPos = ballPos
				BotAI.ballWallCheckTimer = 0
			end
		else
			BotAI.ballWallTimer = math.max(0, BotAI.ballWallTimer - sdt * 2)
			BotAI.ballWallCheckTimer = 0
			BotAI.ballWallLastPos = ballPos
		end

		local nearestChipDist = math.min((ballPos - playerPos).Magnitude, (ballPos - botPos).Magnitude)
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				nearestChipDist = math.min(nearestChipDist, (ballPos - actor.Pos).Magnitude)
			end
		end
		local deadBallTime = (nearestChipDist <= 145) and 0.72 or 1.45
		if touchingWall and BotAI.ballWallEscapeCooldown <= 0 and BotAI.ballWallTimer >= deadBallTime then
			local escapeDir = getDeadBallEscapeDirection(ballPos)
			if escapeDir and escapeDir.Magnitude > 0.01 then
				ballPos = clampToField(ballPos + escapeDir * 7, BALL_R)
				ballVel = escapeDir * (455 * FIELD_SCALE)
				BotAI.ballWallTimer = 0
				BotAI.ballWallCheckTimer = 0
				BotAI.ballWallLastPos = ballPos
				BotAI.ballWallEscapeCooldown = 1.6
				BotAI.ballIdleTimer = 0
				playSfx(SFX.Bounce, 0.045, 0.92)
			end
		elseif isInCornerZone(ballPos, 115) and ballVel.Magnitude < 110 * FIELD_SCALE then
			local inward = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - ballPos)
			ballVel = ballVel + inward * (230 * FIELD_SCALE * sdt)
		elseif touchingWall and BotAI.ballWallTimer > 0.35 then
			ballVel = ballVel + wallInward * (285 * FIELD_SCALE * sdt)
		end

		resolveBallChipOverlaps()
		moveBallSafely(sdt)
		resolveBallChipOverlaps()
		ballVel = ballVel * math.pow(0.34, sdt) -- сильное трение: мяч тяжёлый и быстро останавливается
		if ballVel.Magnitude < 10 * FIELD_SCALE then
			ballVel = Vector2.new(0, 0)
		end
		if ballVel.Magnitude > 1100 * FIELD_SCALE then
			ballVel = unit2(ballVel) * (1100 * FIELD_SCALE)
		end

		-- Штанги
		for _, post in ipairs(POSTS) do
			local diff = ballPos - post
			local dist = diff.Magnitude
			local minDist = BALL_R + 7
			if dist < minDist then
				local n = (dist > 0.001) and (diff / dist) or unit2(ballVel)
				if n.Magnitude < 0.01 then n = Vector2.new((post.X < FIELD_W / 2) and 1 or -1, 0) end
				ballPos = post + n * minDist
				local impactSpeed = bounceBallOffNormal(n, 0.68, 0.92)
				if impactSpeed > 60 * FIELD_SCALE then
					playSfx(SFX.Bounce, 0.04, 1.15)
				end
			end
		end

		-- Верхний и нижний бортики
		if ballPos.Y < BALL_R then
			ballPos = Vector2.new(ballPos.X, BALL_R)
			ballVel = Vector2.new(ballVel.X * 0.94, math.abs(ballVel.Y) * 0.66)
		elseif ballPos.Y > FIELD_H - BALL_R then
			ballPos = Vector2.new(ballPos.X, FIELD_H - BALL_R)
			ballVel = Vector2.new(ballVel.X * 0.94, -math.abs(ballVel.Y) * 0.66)
		end

		-- Скошенные углы-отбойники выталкивают мяч обратно в игру
		collideBallCorner(1, 1)
		collideBallCorner(-1, 1)
		collideBallCorner(1, -1)
		collideBallCorner(-1, -1)

		resolveBallChipOverlaps()

		-- Левые ворота (игрока) / левая стенка
		if ballPos.X < BALL_R then
			if inGoalMouth(ballPos.Y) then
				if ballPos.X < -10 then
					onGoal("Bot")
					return
				end
			else
				ballPos = Vector2.new(BALL_R, ballPos.Y)
				ballVel = Vector2.new(math.abs(ballVel.X) * 0.72, ballVel.Y * 0.94)
			end
		end
		-- Правые ворота (бота) / правая стенка
		if ballPos.X > FIELD_W - BALL_R then
			if inGoalMouth(ballPos.Y) then
				if ballPos.X > FIELD_W + 10 then
					onGoal("player")
					return
				end
			else
				ballPos = Vector2.new(FIELD_W - BALL_R, ballPos.Y)
				ballVel = Vector2.new(-math.abs(ballVel.X) * 0.72, ballVel.Y * 0.94)
			end
		end
		resolveBallChipOverlaps()
	end
	resolveBallChipOverlaps() -- финальная гарантия перед записью повтора и рендером кадра
end

local function playingStep(dt)
	playerStep(dt)
	botStep(dt)
	for _, actor in ipairs(extraActors) do
		extraActorStep(actor, dt)
	end
	resolveChipChip()
	ballStep(dt)
	if state ~= "Playing" then
		return -- забили гол прямо в ballStep
	end

	-- Застрявший мяч (для логики бота)
	local extraNearBall = false
	for _, actor in ipairs(extraActors) do
		if actor.Active and (ballPos - actor.Pos).Magnitude <= 120 then
			extraNearBall = true
			break
		end
	end
	if ballVel.Magnitude < 25 * FIELD_SCALE
		and (ballPos - playerPos).Magnitude > 120
		and (ballPos - botPos).Magnitude > 120
		and not extraNearBall then
		BotAI.ballIdleTimer = BotAI.ballIdleTimer + dt
	else
		BotAI.ballIdleTimer = 0
	end

	-- Таймер матча 2:00
	if not overtime then
		timeLeft = timeLeft - dt
		if timeLeft <= 0 then
			timeLeft = 0
			if playerScore == botScore then
				overtime = true
				showToast("⚡ ЗОЛОТОЙ ГОЛ! Следующий гол — победный!", 2.5)
			else
				updateHUD()
				finishMatch()
				return
			end
		end
	end
	updateHUD()
end

-- Во время баннера «ГОЛ» мяч просто докатывается в сетке
local function ballSettleStep(dt)
	ballPos = ballPos + ballVel * dt
	ballVel = ballVel * math.pow(0.18, dt)
	ballPos = Vector2.new(
		clamp(ballPos.X, -36, FIELD_W + 36),
		clamp(ballPos.Y, BALL_R, FIELD_H - BALL_R)
	)
end

-- Камера поля: Full помещает ВСЁ большое поле в отведённую зону,
-- Close увеличивает поле в 1.7 раза и плавно следит за игроком (чуть-чуть за мячом).
local function updateFieldCamera(dt)
	if not UI.gameFrame or not UI.fieldFrame then
		return
	end
	local viewport = UI.gameFrame.AbsoluteSize
	if viewport.X < 50 or viewport.Y < 50 then
		return
	end

	local availW = viewport.X * 0.92
	local availH = viewport.Y * 0.82
	local areaCenter = Vector2.new(viewport.X * 0.5, viewport.Y * 0.53)
	local ratio = FIELD_W / FIELD_H
	local fitW, fitH
	if availW / availH > ratio then
		fitH = availH
		fitW = fitH * ratio
	else
		fitW = availW
		fitH = fitW / ratio
	end

	local zoom = 1
	local desired = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	if cameraMode == "Close" then
		zoom = 1.7
		local focusPos = Net.GetLocalState()
		desired = focusPos * 0.78 + ballPos * 0.22
		local visibleW = FIELD_W / zoom
		local visibleH = FIELD_H / zoom
		desired = Vector2.new(
			clamp(desired.X, visibleW / 2 - 38 * FIELD_SCALE, FIELD_W - visibleW / 2 + 38 * FIELD_SCALE),
			clamp(desired.Y, visibleH / 2 - 28 * FIELD_SCALE, FIELD_H - visibleH / 2 + 28 * FIELD_SCALE)
		)
		if state == "Countdown" then
			cameraCenter = desired
		else
			cameraCenter = cameraCenter:Lerp(desired, clamp(dt * 5.5, 0, 1))
		end
	else
		cameraCenter = desired
	end

	local fieldWpx = fitW * zoom
	local fieldHpx = fitH * zoom
	local topLeft = Vector2.new(
		areaCenter.X - (cameraCenter.X / FIELD_W) * fieldWpx,
		areaCenter.Y - (cameraCenter.Y / FIELD_H) * fieldHpx
	)
	UI.fieldFrame.Position = UDim2.fromOffset(topLeft.X, topLeft.Y)
	UI.fieldFrame.Size = UDim2.fromOffset(fieldWpx, fieldHpx)
end

local function setDirectionMarker(marker, angle)
	if not marker then
		return
	end
	marker.Position = UDim2.fromScale(
		0.5 + math.cos(angle) * 0.39,
		0.5 + math.sin(angle) * 0.39
	)
	marker.Rotation = math.deg(angle) + 45
end

local function applyDirectionMarkers()
	setDirectionMarker(UI.playerDirMarker, playerFacingAngle)
	setDirectionMarker(UI.botDirMarker, botFacingAngle)
	setDirectionMarker(UI.allyDirMarker, allyActor.Facing)
	setDirectionMarker(UI.ally2DirMarker, ally2Actor.Facing)
	setDirectionMarker(UI.enemy2DirMarker, enemy2Actor.Facing)
	setDirectionMarker(UI.enemy3DirMarker, enemy3Actor.Facing)
end

local function updateDirectionMarkers(dt)
	if playerVel.Magnitude > 18 then
		playerFacingAngle = lerpAngle(playerFacingAngle, math.atan(playerVel.Y, playerVel.X), dt * 14)
	end
	if botVel.Magnitude > 18 then
		botFacingAngle = lerpAngle(botFacingAngle, math.atan(botVel.Y, botVel.X), dt * 14)
	end
	for _, actor in ipairs(extraActors) do
		if actor.Active and actor.Vel.Magnitude > 18 then
			actor.Facing = lerpAngle(actor.Facing, math.atan(actor.Vel.Y, actor.Vel.X), dt * 14)
		end
	end
	applyDirectionMarkers()
end

local function replayStep(dt)
	if not Replay.replayActive then
		return
	end
	Replay.replayTime = Replay.replayTime + dt
	local sourceDuration = Replay.replaySourceEnd - Replay.replaySourceStart
	local sourceT = Replay.replaySourceStart + math.min(sourceDuration, Replay.replayTime * (sourceDuration / Replay.replayDuration))
	while Replay.replayFrameIndex < #Replay.replayFrames and Replay.replayFrames[Replay.replayFrameIndex + 1].t <= sourceT do
		Replay.replayFrameIndex = Replay.replayFrameIndex + 1
	end
	local a = Replay.replayFrames[Replay.replayFrameIndex]
	local b = Replay.replayFrames[math.min(Replay.replayFrameIndex + 1, #Replay.replayFrames)]
	local alpha = 0
	if b.t > a.t then
		alpha = clamp((sourceT - a.t) / (b.t - a.t), 0, 1)
	end
	playerPos = a.p:Lerp(b.p, alpha)
	botPos = a.b:Lerp(b.b, alpha)
	ballPos = a.ball:Lerp(b.ball, alpha)
	ballRotation = a.rot + (b.rot - a.rot) * alpha
	playerFacingAngle = lerpAngle(a.pf, b.pf, alpha)
	botFacingAngle = lerpAngle(a.bf, b.bf, alpha)
	if a.a1 and b.a1 then
		allyActor.Pos = a.a1:Lerp(b.a1, alpha)
		allyActor.Facing = lerpAngle(a.a1f, b.a1f, alpha)
	end
	if a.a2 and b.a2 then
		ally2Actor.Pos = a.a2:Lerp(b.a2, alpha)
		ally2Actor.Facing = lerpAngle(a.a2f, b.a2f, alpha)
	end
	if a.e2 and b.e2 then
		enemy2Actor.Pos = a.e2:Lerp(b.e2, alpha)
		enemy2Actor.Facing = lerpAngle(a.e2f, b.e2f, alpha)
	end
	if a.e3 and b.e3 then
		enemy3Actor.Pos = a.e3:Lerp(b.e3, alpha)
		enemy3Actor.Facing = lerpAngle(a.e3f, b.e3f, alpha)
	end
	applyDirectionMarkers()

	local left = math.max(0, Replay.replayDuration - Replay.replayTime)
	UI.replayTitle.Text = string.format("%s  •  %.1f c",
		replayTitleText(), left)
	UI.replayProgressFill.Size = UDim2.fromScale(clamp(Replay.replayTime / Replay.replayDuration, 0, 1), 1)

	if Replay.replayTime >= Replay.replayDuration then
		endGoalReplay()
	end
end

local function chargeStep(dt)
	if not isCharging then
		return
	end
	chargePower = clamp(chargePower + dt / SETTINGS.CHARGE_TIME, 0, 1) -- полная сила примерно за 0.73 сек
	local shownPower = getEffectivePlayerPower(chargePower) -- и сама шкала растёт мягко, особенно в начале
	UI.powerBarFill.Size = UDim2.fromScale(shownPower, 1)
	if UI.powerPercentLabel then
		UI.powerPercentLabel.Text = tostring(math.floor(shownPower * 100 + 0.5)) .. "%"
	end
	if shownPower < 0.5 then
		UI.powerBarFill.BackgroundColor3 = Color3.fromRGB(46, 204, 113):Lerp(COL.Yellow, shownPower * 2)
	else
		UI.powerBarFill.BackgroundColor3 = COL.Yellow:Lerp(Color3.fromRGB(255, 59, 48), (shownPower - 0.5) * 2)
	end

	-- Стрелка показывает ТОЧНУЮ линию центр фишки -> центр мяча.
	local from
	local dir
	local localChipPos = Net.GetLocalState()
	local inKickRange = (ballPos - localChipPos).Magnitude <= PLAYER_R + BALL_R + PLAYER_KICK_REACH
	if not inKickRange then
		-- Мяч ещё не касается фишки: стрелки нет и рывка нет, шкала просто продолжает расти.
		UI.aimLine.Visible = false
		return
	end
	from = ballPos
	dir = getContactDirection(false) * 100
	local effective = getEffectivePlayerPower(chargePower)
	UI.aimLine.Position = fieldPosToUI(from)
	UI.aimLine.Rotation = math.deg(math.atan(dir.Y, dir.X))
	UI.aimLine.Size = UDim2.new(((55 + effective * 125) * FIELD_SCALE) / FIELD_W, 0, 0, 5)
	UI.aimLine.Visible = true
end

local function findOwnRoom(rooms)
	if type(rooms) ~= "table" then return nil end
	for _, room in ipairs(rooms) do
		if type(room) == "table" and type(room.Players) == "table" then
			for _, info in ipairs(room.Players) do
				if info.UserId == player.UserId then
					return room
				end
			end
		end
	end
	return nil
end

local function callRoom(action, data)
	if not roomRF then
		setNetStatus("Серверная часть не найдена: нужен ROBALL_Online_Server.txt в ServerScriptService")
		return nil
	end
	local ok, result = pcall(function()
		return roomRF:InvokeServer(action, data or {})
	end)
	if not ok or type(result) ~= "table" then
		setNetStatus("Онлайн-сервер не ответил. Проверь Script в ServerScriptService")
		return nil
	end
	if type(result.Rooms) == "table" then
		Net.Rooms = result.Rooms
	end
	if type(result.Room) == "table" then
		Net.Room = result.Room
		Net.InRoom = true
	elseif action == "Leave" and result.Ok then
		Net.Room = nil
		Net.InRoom = false
		Net.InMatch = false
	elseif type(result.Rooms) == "table" then
		local own = findOwnRoom(result.Rooms)
		Net.Room = own
		Net.InRoom = own ~= nil
	end
	if result.Message then
		setNetStatus(result.Message)
	end
	if UI.renderOnlineRooms then
		UI.renderOnlineRooms()
	end
	return result
end

local function setSlotState(slot, pos, vel, facing)
	if slot == 1 then
		playerPos, playerVel = pos, vel
		if facing then playerFacingAngle = facing end
	elseif slot == 2 then
		botPos, botVel = pos, vel
		if facing then botFacingAngle = facing end
	elseif slot == 3 then
		allyActor.Pos, allyActor.Vel = pos, vel
		if facing then allyActor.Facing = facing end
	elseif slot == 4 then
		enemy2Actor.Pos, enemy2Actor.Vel = pos, vel
		if facing then enemy2Actor.Facing = facing end
	elseif slot == 5 then
		ally2Actor.Pos, ally2Actor.Vel = pos, vel
		if facing then ally2Actor.Facing = facing end
	elseif slot == 6 then
		enemy3Actor.Pos, enemy3Actor.Vel = pos, vel
		if facing then enemy3Actor.Facing = facing end
	end
end

local function snapToTargets()
	local targets = Net.Targets
	if not targets then return end
	if targets.Ball then
		ballPos = targets.Ball
		ballVel = targets.BallVel or Vector2.new(0, 0)
		if targets.BallRot then ballRotation = targets.BallRot end
	end
	for slot, info in pairs(targets.Slots or {}) do
		setSlotState(slot, info.Pos, info.Vel, info.Facing)
	end
end

local function lerpSlot(slot, target, alpha)
	if slot == 1 then
		playerPos = playerPos:Lerp(target.Pos, alpha)
		playerVel = target.Vel
		if target.Facing then playerFacingAngle = lerpAngle(playerFacingAngle, target.Facing, alpha) end
	elseif slot == 2 then
		botPos = botPos:Lerp(target.Pos, alpha)
		botVel = target.Vel
		if target.Facing then botFacingAngle = lerpAngle(botFacingAngle, target.Facing, alpha) end
	elseif slot == 3 then
		allyActor.Pos = allyActor.Pos:Lerp(target.Pos, alpha)
		allyActor.Vel = target.Vel
		if target.Facing then allyActor.Facing = lerpAngle(allyActor.Facing, target.Facing, alpha) end
	elseif slot == 4 then
		enemy2Actor.Pos = enemy2Actor.Pos:Lerp(target.Pos, alpha)
		enemy2Actor.Vel = target.Vel
		if target.Facing then enemy2Actor.Facing = lerpAngle(enemy2Actor.Facing, target.Facing, alpha) end
	elseif slot == 5 then
		ally2Actor.Pos = ally2Actor.Pos:Lerp(target.Pos, alpha)
		ally2Actor.Vel = target.Vel
		if target.Facing then ally2Actor.Facing = lerpAngle(ally2Actor.Facing, target.Facing, alpha) end
	elseif slot == 6 then
		enemy3Actor.Pos = enemy3Actor.Pos:Lerp(target.Pos, alpha)
		enemy3Actor.Vel = target.Vel
		if target.Facing then enemy3Actor.Facing = lerpAngle(enemy3Actor.Facing, target.Facing, alpha) end
	end
end

-- Лёгкая визуальная страховка онлайна: двигаем только мяч, если интерполяция
-- на кадр посадила его внутрь фишки. Полный офлайн-решатель здесь не нужен:
-- он на каждом кадре толкал сами фишки/скорости и мог создавать тряску и
-- лишнюю нагрузку на телефоне. Позиции людей здесь не меняем вообще.
local function softSeparateOnlineBall()
	local minDist = PLAYER_R + BALL_R + 0.15
	for _pass = 1, 2 do
		local function pushFrom(chipPos)
			local diff = ballPos - chipPos
			local dist = diff.Magnitude
			if dist < minDist then
				local normal = (dist > 0.01) and (diff / dist) or Vector2.new(1, 0)
				ballPos = chipPos + normal * minDist
			end
		end
		pushFrom(playerPos)
		pushFrom(botPos)
		for _, actor in ipairs(extraActors) do
			if actor.Active then
				pushFrom(actor.Pos)
			end
		end
	end
	keepBallInsideForSolver()
end

local function applySnapshot(data)
	if not Net.InMatch or type(data) ~= "table" then return end
	-- UnreliableRemoteEvent не гарантирует порядок: старые пакеты игнорируем.
	if type(data.Seq) == "number" then
		if data.Seq <= (Net.LastSnapshotSeq or 0) then return end
		Net.LastSnapshotSeq = data.Seq
	end

	-- Сервер присылает своё время снимка. Компенсируем возраст пакета
	-- экстраполяцией скоростей: клиент рисует не прошлую картинку, а оценку
	-- текущего серверного положения. Вне Playing сервер позы не интегрирует,
	-- поэтому там экстраполяцию не включаем.
	local serverNow = workspace:GetServerTimeNow()
	local serverTime = tonumber(data.ServerTime) or serverNow
	local snapshotAge = (data.State == "Playing") and clamp(serverNow - serverTime, 0, 0.22) or 0
	Net.LastSnapshotServerTime = serverTime
	Net.SnapshotAge = snapshotAge

	Net.SnapshotState = data.State or Net.SnapshotState
	if type(data.TimeLeft) == "number" then timeLeft = data.TimeLeft end
	if type(data.BlueScore) == "number" then playerScore = math.floor(data.BlueScore) end
	if type(data.RedScore) == "number" then botScore = math.floor(data.RedScore) end
	if type(data.BlueShots) == "number" then shotsPlayer = math.floor(data.BlueShots) end
	if type(data.RedShots) == "number" then shotsBot = math.floor(data.RedShots) end
	overtime = data.Overtime == true
	if type(data.CountdownEndsAt) == "number" then Net.CountdownEndsAt = data.CountdownEndsAt end

	local targets = Net.Targets or { Slots = {} }
	targets.Slots = targets.Slots or {}
	-- Сырой снимок без компенсации возраста кладём во временной буфер.
	-- Net.Step нарисует чужих игроков и мяч между соседними снимками:
	-- сетевой джиттер перестанет превращаться в быструю тряску фишек.
	local bufferEntry = { Time = serverTime, Slots = {} }
	if type(data.Ball) == "table" then
		-- Компактный формат сервера: {X, Y, VX, VY, Rot}.
		local rawBallPos = Vector2.new(tonumber(data.Ball[1] or data.Ball.X) or ballPos.X, tonumber(data.Ball[2] or data.Ball.Y) or ballPos.Y)
		local rawBallVel = Vector2.new(tonumber(data.Ball[3] or data.Ball.VX) or 0, tonumber(data.Ball[4] or data.Ball.VY) or 0)
		local ballTravel = snapshotAge
		if snapshotAge > 0 then
			-- Интеграл скорости с серверным трением 0.34^dt, а не грубый рывок Vel * age.
			ballTravel = (1 - math.pow(0.34, snapshotAge)) / 1.0788
		end
		targets.Ball = rawBallPos + rawBallVel * ballTravel
		targets.BallVel = rawBallVel * math.pow(0.34, snapshotAge)
		targets.BallRot = tonumber(data.Ball[5] or data.Ball.Rot)
		bufferEntry.Ball = rawBallPos
		bufferEntry.BallVel = rawBallVel
		bufferEntry.BallRot = targets.BallRot
	end

	local localAuth = nil
	if type(data.Players) == "table" then
		for slot, info in pairs(data.Players) do
			local slotNumber = tonumber(slot)
			if slotNumber and type(info) == "table" then
				-- Компактный формат сервера: {X, Y, VX, VY, Facing}.
				local rawPos = Vector2.new(tonumber(info[1] or info.X) or 0, tonumber(info[2] or info.Y) or 0)
				local rawVel = Vector2.new(tonumber(info[3] or info.VX) or 0, tonumber(info[4] or info.VY) or 0)
				local targetPos = rawPos + rawVel * snapshotAge
				targetPos = clampToField(targetPos, PLAYER_R)
				local target = {
					Pos = targetPos,
					Vel = rawVel,
					Facing = tonumber(info[5] or info.F),
				}
				targets.Slots[slotNumber] = target
				bufferEntry.Slots[slotNumber] = {
					Pos = clampToField(rawPos, PLAYER_R),
					Vel = rawVel,
					Facing = target.Facing,
				}
				if slotNumber == Net.LocalSlot then
					localAuth = target
				end
			end
		end
	end
	Net.Targets = targets

	local shouldSnap = data.Snap == true or not Net.HasSnapshot
	if not shouldSnap and targets.Ball and (targets.Ball - ballPos).Magnitude > 320 * FIELD_SCALE then
		shouldSnap = true
	end
	Net.HasSnapshot = true
	if shouldSnap then
		Net.SnapshotBuffer = { bufferEntry }
	else
		local buffer = Net.SnapshotBuffer or {}
		table.insert(buffer, bufferEntry)
		while #buffer > 2 and buffer[2].Time <= serverTime - 0.45 do
			table.remove(buffer, 1)
		end
		Net.SnapshotBuffer = buffer
	end
	Net.LocalAuth = localAuth

	if shouldSnap then
		snapToTargets()
	elseif localAuth then
		-- В момент пакета свою фишку больше не дёргаем мгновенной коррекцией.
		-- Net.Step ниже каждый RenderStepped плавно и с ограничением скорости
		-- подтянет предсказание к Net.LocalAuth. Сюда оставлен только аварийный
		-- перенос при огромной ошибке (столкновение, телепорт, потеря пакетов).
		local currentPos = Net.GetLocalState()
		if (localAuth.Pos - currentPos).Magnitude > 170 * FIELD_SCALE then
			setSlotState(Net.LocalSlot, localAuth.Pos, localAuth.Vel, localAuth.Facing)
		end
	end

	if data.State == "Countdown" then
		state = "Countdown"
		UI.countdownLabel.Visible = true
	elseif data.State == "Playing" then
		state = "Playing"
		UI.countdownLabel.Visible = false
	elseif data.State == "Goal" then
		state = "Goal"
		UI.countdownLabel.Visible = false
	elseif data.State == "Finished" then
		state = "Finished"
		UI.countdownLabel.Visible = false
	end
	updateHUD()
end

local function onMatchEvent(data)
	if type(data) ~= "table" or type(data.Type) ~= "string" then return end
	if data.Type == "MatchStarting" then
		startOnlineMatch(data)
	elseif data.Type == "Playing" then
		if Net.InMatch then
			state = "Playing"
			UI.countdownLabel.Visible = false
			playSfx(SFX.Click, 0.10, 1.35)
		end
	elseif data.Type == "Kickoff" then
		if Net.InMatch then
			state = "Countdown"
			Net.CountdownEndsAt = tonumber(data.CountdownEndsAt) or (workspace:GetServerTimeNow() + 2)
			UI.goalBanner.Visible = false
			UI.countdownLabel.Visible = true
		end
	elseif data.Type == "Goal" then
		if not Net.InMatch then return end
		if type(data.BlueScore) == "number" then playerScore = math.floor(data.BlueScore) end
		if type(data.RedScore) == "number" then botScore = math.floor(data.RedScore) end
		state = "Goal"
		cancelCharge()
		local scorerTeam = data.ScorerTeam
		UI.goalBanner.Text = (scorerTeam == Net.LocalTeam) and "⚽ ГОЛ ТВОЕЙ КОМАНДЫ!" or "⚽ ГОЛ СОПЕРНИКА..."
		UI.goalBanner.TextColor3 = (scorerTeam == Net.LocalTeam) and COL.Yellow or Color3.fromRGB(255, 110, 100)
		UI.goalBanner.Visible = true
		UI.goalBannerScale.Scale = 0.4
		TweenService:Create(UI.goalBannerScale, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Scale = 1 }):Play()
		playSfx(SFX.Goal, 0.42, 1, 6)
		spawnGoalFireworks((scorerTeam == "Blue") and "player" or "Bot")
		Net.GoalToken = Net.GoalToken + 1
		local myGoal = Net.GoalToken
		task.delay(1.6, function()
			if Net.GoalToken == myGoal then UI.goalBanner.Visible = false end
		end)
		updateHUD()
	elseif data.Type == "Kick" then
		local p = clamp(tonumber(data.Power) or 0.4, 0, 1)
		playSfx(SFX.Kick, 0.18 + p * 0.18, 0.95 + p * 0.10)
		pulseBall()
	elseif data.Type == "GoldenGoal" then
		overtime = true
		showToast("⚡ ЗОЛОТОЙ ГОЛ! Следующий гол — победный!", 2.5)
		updateHUD()
	elseif data.Type == "Finished" then
		if type(data.BlueScore) == "number" then playerScore = math.floor(data.BlueScore) end
		if type(data.RedScore) == "number" then botScore = math.floor(data.RedScore) end
		if type(data.BlueShots) == "number" then shotsPlayer = math.floor(data.BlueShots) end
		if type(data.RedShots) == "number" then shotsBot = math.floor(data.RedShots) end
		Net.WinnerTeam = data.WinnerTeam
		Net.WinnerReason = data.Reason
		Net.SnapshotState = "Finished"
		finishMatch()
	elseif data.Type == "Aborted" then
		Net.InMatch = false
		Net.Targets = nil
		Net.SnapshotBuffer = {}
		Net.LocalAuth = nil
		Net.HasSnapshot = false
		Net.LastSnapshotSeq = 0
		goToMenu()
		if UI.setMenuTab then UI.setMenuTab("Online") end
		setNetStatus(data.Message or "Матч остановлен: игрок вышел из комнаты")
		Net.RefreshRooms()
	elseif data.Type == "Toast" then
		if data.Message then showToast(data.Message, 2) end
	end
end

Net.Step = function(dt)
	if not Net.InMatch then return end

	-- 1) Сначала читаем свежий ввод именно этого кадра. Своя фишка ниже
	-- сдвинется сразу, не дожидаясь путешествия пакета на сервер и обратно.
	local move = Vector2.new(0, 0)
	if state == "Playing" then
		if UI.joystickActive and UI.joystickVector and UI.joystickVector.Magnitude > 0.01 then
			move = UI.joystickVector
		elseif touchActive then
			local localPos = Net.GetLocalState()
			local toTouch = touchFieldPos - localPos
			if toTouch.Magnitude > 12 then
				move = unit2(toTouch) * clamp(toTouch.Magnitude / (95 * FIELD_SCALE), 0.25, 1)
			end
		elseif not UserInputService:GetFocusedTextBox() then
			if keysDown[Enum.KeyCode.W] or keysDown[Enum.KeyCode.Up] then move = move + Vector2.new(0, -1) end
			if keysDown[Enum.KeyCode.S] or keysDown[Enum.KeyCode.Down] then move = move + Vector2.new(0, 1) end
			if keysDown[Enum.KeyCode.A] or keysDown[Enum.KeyCode.Left] then move = move + Vector2.new(-1, 0) end
			if keysDown[Enum.KeyCode.D] or keysDown[Enum.KeyCode.Right] then move = move + Vector2.new(1, 0) end
			if move.Magnitude > 1 then move = unit2(move) end
		end
	end
	local sprint = keysDown[Enum.KeyCode.LeftShift] or keysDown[Enum.KeyCode.RightShift]

	-- 2) Движение шлём UnreliableRemoteEvent-ом: серверу нужен только самый
	-- свежий вектор, старые пакеты не должны стоять в надёжной очереди.
	-- Клавиатуре хватает 30 Гц, а аналоговый палец меняет вектор непрерывно,
	-- поэтому активный джойстик шлём до 60 Гц — сервер меньше отстаёт от руки.
	-- Seq защищает от смены порядка. Удар идёт отдельным надёжным каналом.
	local inputKey = string.format("%.2f|%.2f|%s|%s|%s", move.X, move.Y, tostring(sprint), tostring(isCharging), state)
	Net.InputTimer = Net.InputTimer - dt
	if matchInputRemote and Net.InputTimer <= 0 and (state == "Playing" or inputKey ~= Net.LastInputKey) then
		Net.InputSeq = (Net.InputSeq or 0) + 1
		matchInputRemote:FireServer({
			Seq = Net.InputSeq,
			MoveX = move.X,
			MoveY = move.Y,
			Sprint = sprint == true,
			Charging = isCharging == true,
		})
		Net.InputTimer = (UI.joystickActive or touchActive) and (1 / 60) or (1 / 30)
		Net.LastInputKey = inputKey
	end

	-- 3) Локальное предсказание своей фишки каждый RenderStepped.
	-- Скорость здесь НАКАПЛИВАЕТСЯ из предыдущего предсказания; серверный
	-- снапшот в applySnapshot делает только мягкую коррекцию ошибки.
	if state == "Playing" and Net.LocalSlot >= 1 and Net.LocalSlot <= 6 then
		local localPos, localVel = Net.GetLocalState()
		local maxSpeed = ((sprint and SETTINGS.PLAYER_SPRINT) or SETTINGS.PLAYER_SPEED) * FIELD_SCALE
		if isCharging then
			maxSpeed = maxSpeed * 0.72
		end
		local desired = move * maxSpeed
		local acceleration = 820 * FIELD_SCALE
		if desired.Magnitude < 1 then
			acceleration = 980 * FIELD_SCALE
		elseif localVel.Magnitude > 20 * FIELD_SCALE and localVel:Dot(desired) < 0 then
			acceleration = 900 * FIELD_SCALE
		end
		localVel = approachVelocity(localVel, desired, acceleration, dt)
		if desired.Magnitude < 1 and localVel.Magnitude < 5 then
			localVel = Vector2.new(0, 0)
		end
		localPos = localPos + localVel * dt
		localPos, localVel = applyFieldClamp(localPos, localVel, PLAYER_R)
		setSlotState(Net.LocalSlot, localPos, localVel, nil)
	end

	-- 3.5) Коррекцию своей фишки растягиваем по RenderStepped-кадрам.
	-- Раньше ошибка чинилась одним Lerp-шагом в момент каждого пакета 30 Гц:
	-- при обычном сетевом джиттере это выглядело как мелкая быстрая тряска.
	-- Теперь за кадр разрешён ограниченный сдвиг, а ошибка до 9 px игнорируется.
	local auth = Net.LocalAuth
	if auth and Net.LocalSlot >= 1 and Net.LocalSlot <= 6 then
		if state == "Playing" then
			auth.Pos = clampToField(auth.Pos + auth.Vel * dt, PLAYER_R)
		end
		local currentPos, currentVel = Net.GetLocalState()
		local positionError = auth.Pos - currentPos
		local errorDist = positionError.Magnitude
		if errorDist > 170 * FIELD_SCALE then
			setSlotState(Net.LocalSlot, auth.Pos, auth.Vel, auth.Facing)
		elseif errorDist > 9 * FIELD_SCALE then
			local requested = errorDist * (1 - math.exp(-7 * dt))
			local maxCorrection = 280 * FIELD_SCALE * dt
			local correctedPos = currentPos + unit2(positionError) * math.min(requested, maxCorrection)
			local correctedVel = currentVel
			if errorDist > 48 * FIELD_SCALE then
				correctedVel = currentVel:Lerp(auth.Vel, 1 - math.exp(-5 * dt))
			end
			setSlotState(Net.LocalSlot, correctedPos, correctedVel, nil)
		end
	end

	-- 4) Чужие фишки и мяч: интерполяция между двумя серверными снимками.
	-- Рендер-время специально на 92 мс позади серверного. Это крошечная
	-- фиксированная задержка только для чужих объектов, зато движение между
	-- пакетами становится равномерным, а джиттер доставки не дёргает людей.
	-- Своя фишка сюда не попадает: она по-прежнему предсказывается мгновенно.
	local buffer = Net.SnapshotBuffer
	if buffer and #buffer > 0 then
		local renderTime = workspace:GetServerTimeNow() - 0.092
		while #buffer > 2 and buffer[2].Time <= renderTime do
			table.remove(buffer, 1)
		end
		local latest = buffer[#buffer]
		local fromSnap = buffer[1]
		local toSnap = latest
		for i = #buffer - 1, 1, -1 do
			if buffer[i].Time <= renderTime then
				fromSnap = buffer[i]
				toSnap = buffer[i + 1]
				break
			end
		end
		local span = toSnap.Time - fromSnap.Time
		local alpha = (span > 0.0001) and clamp((renderTime - fromSnap.Time) / span, 0, 1) or 1
		local extrapolate = 0
		if renderTime > latest.Time then
			-- Пакет задержался: коротко и ограниченно продолжаем последний снимок,
			-- вместо замирания или резкого прыжка на следующем пакете.
			extrapolate = math.min(renderTime - latest.Time, 0.08)
			fromSnap = latest
			toSnap = latest
			alpha = 1
		end

		for slot = 1, 6 do
			if slot ~= Net.LocalSlot then
				local aInfo = fromSnap.Slots and fromSnap.Slots[slot]
				local bInfo = toSnap.Slots and toSnap.Slots[slot]
				if aInfo and bInfo then
					local smoothPos
					if extrapolate > 0 and state == "Playing" then
						smoothPos = clampToField(bInfo.Pos + bInfo.Vel * extrapolate, PLAYER_R)
					else
						smoothPos = aInfo.Pos:Lerp(bInfo.Pos, alpha)
					end
					local smoothFacing = bInfo.Facing
					if aInfo.Facing and bInfo.Facing then
						smoothFacing = lerpAngle(aInfo.Facing, bInfo.Facing, alpha)
					end
					local smoothVel = bInfo.Vel
					if extrapolate <= 0 and span > 0.0001 and span <= 0.15 then
						-- Скорость именно рисуемого отрезка: маркер направления
						-- не дёргается от мгновенного серверного Vel каждого пакета.
						smoothVel = (bInfo.Pos - aInfo.Pos) / span
					end
					setSlotState(slot, smoothPos, smoothVel, smoothFacing)
				elseif bInfo then
					setSlotState(slot, bInfo.Pos, bInfo.Vel, bInfo.Facing)
				end
			end
		end

		if fromSnap.Ball and toSnap.Ball then
			if extrapolate > 0 and state == "Playing" and toSnap.BallVel then
				local travel = (1 - math.pow(0.34, extrapolate)) / 1.0788
				ballPos = toSnap.Ball + toSnap.BallVel * travel
				ballVel = toSnap.BallVel * math.pow(0.34, extrapolate)
				if ballVel.Magnitude < 10 * FIELD_SCALE then
					ballVel = Vector2.new(0, 0)
				end
				if inGoalMouth(ballPos.Y) then
					ballPos = Vector2.new(ballPos.X, clamp(ballPos.Y, BALL_R, FIELD_H - BALL_R))
				else
					ballPos = clampToField(ballPos, BALL_R)
				end
			else
				ballPos = fromSnap.Ball:Lerp(toSnap.Ball, alpha)
				if fromSnap.BallVel and toSnap.BallVel then
					ballVel = fromSnap.BallVel:Lerp(toSnap.BallVel, alpha)
				elseif toSnap.BallVel then
					ballVel = toSnap.BallVel
				end
			end
		elseif toSnap.Ball then
			ballPos = toSnap.Ball
			ballVel = toSnap.BallVel or ballVel
		end
	end

	if state == "Playing" then
		-- Только лёгкая страховка мяча. Людей здесь не толкаем: полный решатель
		-- на каждом кадре онлайна мог сам создавать тряску и грузить телефон.
		softSeparateOnlineBall()
	end

	if state == "Countdown" then
		local remain = Net.CountdownEndsAt - workspace:GetServerTimeNow()
		UI.countdownLabel.Visible = true
		UI.countdownLabel.Text = (remain > 0.15) and tostring(math.ceil(remain)) or "⚽ ИГРА!"
	elseif state == "Playing" then
		UI.countdownLabel.Visible = false
	end
end

task.spawn(function()
	local folder = ReplicatedStorage:WaitForChild("ROBALL_Online", 12)
	if not folder then
		setNetStatus("Серверная часть не найдена: вставь ROBALL_Online_Server.txt в ServerScriptService")
		return
	end
	roomRF = folder:WaitForChild("RoomRF", 8)
	roomUpdateRemote = folder:WaitForChild("RoomUpdate", 8)
	matchInputRemote = folder:WaitForChild("MatchInput", 8)
	matchKickRemote = folder:WaitForChild("MatchKick", 8)
	matchEventRemote = folder:WaitForChild("MatchEvent", 8)
	matchSnapshotRemote = folder:WaitForChild("MatchSnapshot", 8)
	if not roomRF or not roomUpdateRemote or not matchInputRemote or not matchKickRemote or not matchEventRemote or not matchSnapshotRemote then
		setNetStatus("Онлайн-сервер найден, но RemoteEvent-ы созданы не полностью")
		return
	end
	Net.Connected = true
	Net.Bound = true
	roomUpdateRemote.OnClientEvent:Connect(function(data)
		if type(data) ~= "table" or type(data.Rooms) ~= "table" then return end
		Net.Rooms = data.Rooms
		local own = findOwnRoom(data.Rooms)
		Net.Room = own
		Net.InRoom = own ~= nil
		if UI.renderOnlineRooms then UI.renderOnlineRooms() end
		if Net.InMatch and state == "Finished" and own and own.Status == "Finished" then
			finishMatch()
		end
	end)
	matchEventRemote.OnClientEvent:Connect(onMatchEvent)
	matchSnapshotRemote.OnClientEvent:Connect(applySnapshot)
	setNetStatus("🟢 Онлайн подключён • комнаты видны игрокам этого сервера Roblox")
	Net.RefreshRooms()
	while true do
		task.wait(5)
		if UI.menuFrame.Visible and Net.MenuTab == "Online" and not Net.InMatch then
			Net.RefreshRooms()
		end
	end
end)
end -- онлайн-клиент

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.Keyboard then
		keysDown[input.KeyCode] = true
		if input.KeyCode == Enum.KeyCode.P and not gameProcessed then
			if state == "Playing" or state == "Paused" then
				togglePause(false)
			end
		end
		if input.KeyCode == Enum.KeyCode.Space and not gameProcessed then
			if state == "Replay" then
				endGoalReplay()
			elseif state == "Goal" and not Net.InMatch then
				Replay.replaySkipRequested = true
			elseif state == "Playing" then
				-- Быстрый физический удар с пробела (бонус во время игры)
				isCharging = true
				chargeAimToGoal = true
				chargePower = 0.50
				releaseKick()
			end
		end
	end

	if input.UserInputType == Enum.UserInputType.MouseButton1 and state == "Playing" and not gameProcessed then
		local mousePos = mouseGuiPos()
		-- Заряжаем удар ТОЛЬКО по клику внутри поля (клики по кнопкам не трогаем)
		if isScreenPosInsideField(mousePos) then
			isCharging = true
			chargeAimToGoal = false
			chargePower = 0 -- сила всегда начинается с нуля и растёт медленно
			UI.powerBarBg.Visible = true
		end
	end

	if input.UserInputType == Enum.UserInputType.Touch and UI.gameFrame.Visible then
		local p = toGuiSpace(Vector2.new(input.Position.X, input.Position.Y))
		-- Джойстик обрабатываем первым и запоминаем именно этот палец:
		-- второй палец в этот момент может спокойно держать кнопку мяча.
		if (state == "Playing" or state == "Countdown") and UI.beginJoystick and UI.beginJoystick(input, p) then
			return
		end
		if state == "Playing" and not gameProcessed and isScreenPosInsideField(p) then
			touchActive = true
			UI.fieldTouchInput = input
			touchFieldPos = screenToField(p)
		end
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch then
		local p = toGuiSpace(Vector2.new(input.Position.X, input.Position.Y))
		if UI.moveJoystick and UI.moveJoystick(input, p) then
			return
		end
		if touchActive and UI.fieldTouchInput == input then
			touchFieldPos = screenToField(p)
		end
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Keyboard then
		keysDown[input.KeyCode] = false
	end
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		if isCharging and not chargeAimToGoal then
			releaseKick()
		end
	end
	if input.UserInputType == Enum.UserInputType.Touch then
		if UI.kickTouchInput == input then
			UI.kickTouchInput = nil
			if isCharging and chargeAimToGoal then
				releaseKick()
			end
		end
		local wasJoystick = UI.endJoystick and UI.endJoystick(input)
		if not wasJoystick and UI.fieldTouchInput == input then
			touchActive = false
			UI.fieldTouchInput = nil
		end
	end
end)

-- Если окно игры потеряло фокус — не залипаем с зажатыми клавишами

	keysDown = {}
	touchActive = false
	UI.fieldTouchInput = nil
	UI.kickTouchInput = nil
	if UI.resetJoystick then UI.resetJoystick() end
	if isCharging then
		cancelCharge()
	end
end)

RunService.RenderStepped:Connect(function(dt)
	local netDt = math.min(dt, 0.05)
	if dt > 1 / 30 then
		dt = 1 / 30
	end

	if UI.menuFrame.Visible and UI.rainbowGradient then
		UI.rainbowGradient.Offset = Vector2.new((os.clock() * 0.20) % 1, 0)
		UI.rainbowGradient.Rotation = math.sin(os.clock() * 0.7) * 6
	end

	if not UI.gameFrame.Visible then
		return
	end

	if UI.stepJoystick then UI.stepJoystick(netDt) end

	if Net.InMatch then
		-- Онлайн-предсказание получает реальный dt до 50 мс: на слабом FPS оно не отстаёт от сервера из-за общего капа 1/30.
		if Net.Step then Net.Step(netDt) end
		updateDirectionMarkers(dt)
		if state == "Playing" then chargeStep(dt) end
	elseif state == "Playing" then
		playingStep(dt)
		updateDirectionMarkers(dt)
		recordReplayFrame(dt)
		chargeStep(dt)
	elseif state == "Goal" then
		ballSettleStep(dt)
		recordReplayFrame(dt)
	elseif state == "Replay" then
		replayStep(dt)
	elseif state == "Countdown" then
		-- Стоим на местах, ждём свистка
	end

	updateFieldCamera(dt)

	-- Мяч катится — крутим его тело (угол пишется в повтор без скачков 360->0)
	if state == "Playing" or state == "Goal" then
		ballRotation = ballRotation + ballVel.Magnitude * dt * 0.45
	end

	render()
end)
