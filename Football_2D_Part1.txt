--// Маленькие помощники \\--
local function clamp(v, a, b)
	if v < a then return a end
	if v > b then return b end
	return v
end

local function unit2(v)
	local m = v.Magnitude
	if m < 0.0001 then
		return Vector2.new(0, 0)
	end
	return v / m
end

-- Кратчайшая плавная интерполяция угла (для поворота персонажей и повтора)
local function lerpAngle(a, b, alpha)
	local diff = (b - a + math.pi) % (math.pi * 2) - math.pi
	return a + diff * clamp(alpha, 0, 1)
end

-- Плавная физика фишки: скорость меняется не мгновенно, а с ускорением.
-- Из-за этого есть разгон, инерция, скольжение при остановке и мягкие повороты.
local function approachVelocity(current, desired, acceleration, dt)
	local delta = desired - current
	local maxChange = acceleration * dt
	if delta.Magnitude <= maxChange then
		return desired
	end
	return current + unit2(delta) * maxChange
end

-- Геометрия поля со скошенными углами: круг радиуса r не может забиться в мёртвый угол.
local function clampToField(pos, r)
	local p = Vector2.new(
		clamp(pos.X, r, FIELD_W - r),
		clamp(pos.Y, r, FIELD_H - r)
	)
	local need = CORNER_CUT + r * math.sqrt(2)
	local tl = p.X + p.Y
	if tl < need then
		local add = (need - tl) / 2
		p = Vector2.new(p.X + add, p.Y + add)
	end
	local tr = (FIELD_W - p.X) + p.Y
	if tr < need then
		local add = (need - tr) / 2
		p = Vector2.new(p.X - add, p.Y + add)
	end
	local bl = p.X + (FIELD_H - p.Y)
	if bl < need then
		local add = (need - bl) / 2
		p = Vector2.new(p.X + add, p.Y - add)
	end
	local br = (FIELD_W - p.X) + (FIELD_H - p.Y)
	if br < need then
		local add = (need - br) / 2
		p = Vector2.new(p.X - add, p.Y - add)
	end
	return p
end

local function isInCornerZone(pos, pad)
	return (pos.X < pad or pos.X > FIELD_W - pad)
		and (pos.Y < pad or pos.Y > FIELD_H - pad)
end

local function create(className, props, parent)
	local inst = Instance.new(className)
	for k, v in pairs(props) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function addCorner(obj, scale, offset)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(scale or 0, offset or 0)
	c.Parent = obj
	return c
end

local function addStroke(obj, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color
	s.Thickness = thickness or 2
	s.Transparency = transparency or 0
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = obj
	return s
end

local function addGradient(obj, c1, c2, rotation)
	local g = Instance.new("UIGradient")
	g.Color = ColorSequence.new(c1, c2)
	g.Rotation = rotation or 90
	g.Parent = obj
	return g
end

-- Безопасные звуки: если какого-то файла в игре нет — просто тишина, игра не ломается
local function playSfx(assetId, volume, pitch, life)
	local ok = pcall(function()
		local s = Instance.new("Sound")
		s.SoundId = assetId
		s.Volume = volume or 0.5
		s.PlaybackSpeed = pitch or 1
		s.Parent = SoundService
		s:Play()
		Debris:AddItem(s, life or 2)
	end)
	return ok
end

local function makeButton(parent, props)
	local base = {
		BackgroundColor3 = COL.GreenBtn,
		Text = "",
		TextColor3 = COL.White,
		Font = Enum.Font.GothamBold,
		TextSize = 20,
		AutoButtonColor = false,
		BorderSizePixel = 0,
	}
	for k, v in pairs(props) do
		base[k] = v
	end
	local btn = create("TextButton", base, parent)
	local scale = Instance.new("UIScale")
	scale.Parent = btn
	btn.MouseEnter:Connect(function()
		TweenService:Create(scale, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Scale = 1.05 }):Play()
	end)
	btn.MouseLeave:Connect(function()
		TweenService:Create(scale, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Scale = 1 }):Play()
	end)
	btn.MouseButton1Click:Connect(function()
		playSfx(SFX.Click, 0.12, 1.15)
	end)
	return btn
end

-- Конвертеры координат: виртуальное поле 1000x600 <-> экран
-- ВНИМАНИЕ: GetMouseLocation() и InputObject.Position считают   от   верхнего   угла   окна (с топбаром),
-- а AbsolutePosition у GUI — уже БЕЗ него. Вычитаем GuiInset, иначе прицел съедет на ~36 пикселей по Y.
local function toGuiSpace(rawPos)
	local topLeftInset = GuiService:GetGuiInset()
	return rawPos - topLeftInset
end

local function mouseGuiPos()
	return toGuiSpace(UserInputService:GetMouseLocation())
end

local function fieldPosToUI(pos)

end

local function fieldSizeToUI(w, h)
	return UDim2.fromScale(w / FIELD_W, h / FIELD_H)
end

local function screenToField(screenPos)
	local ap = fieldFrame.AbsolutePosition
	local as = fieldFrame.AbsoluteSize
	if as.X < 1 or as.Y < 1 then
		return ballPos
	end
	local rel = screenPos - ap
	return Vector2.new(rel.X / as.X * FIELD_W, rel.Y / as.Y * FIELD_H)
end

local function isScreenPosInsideField(screenPos)
	-- В крупном плане поле увеличено и частично за кадром: удар принимаем по всей
	-- игровой зоне ниже верхней панели (сама панель с кнопками не трогается).
	if cameraMode == "Close" then
		local gs = gameFrame.AbsoluteSize
		return screenPos.Y > gs.Y * 0.108 and screenPos.Y < gs.Y * 0.99
	end
	local ap = fieldFrame.AbsolutePosition
	local as = fieldFrame.AbsoluteSize
	return screenPos.X >= ap.X and screenPos.X <= ap.X + as.X
		and screenPos.Y >= ap.Y and screenPos.Y <= ap.Y + as.Y
end

function showToast(text, duration)
	toastToken = toastToken + 1
	local my = toastToken
	toastLabel.Text = text
	toastLabel.Visible = true
	task.delay(duration or 2, function()
		if toastToken == my then
			toastLabel.Visible = false
		end
	end)
end

function updateHUD()
	scoreLeftLabel.Text = "ВЫ  " .. tostring(playerScore)
	scoreRightLabel.Text = tostring(botScore) .. "  БОТ"
	if overtime then
		timerLabel.Text = "ЗОЛОТОЙ ГОЛ"
		timerLabel.TextColor3 = COL.Yellow
	else
		local t = math.max(0, math.ceil(timeLeft))
		local m = math.floor(t / 60)
		local s = t % 60
		timerLabel.Text = string.format("%d:%02d", m, s)
		if timeLeft <= 10 then
			timerLabel.TextColor3 = Color3.fromRGB(255, 80, 70)
		else
			timerLabel.TextColor3 = COL.White
		end
	end
end

local function cancelCharge()
	isCharging = false
	chargePower = 0
	chargeAimToGoal = false
	powerBarBg.Visible = false
	aimLine.Visible = false
end

-- Расстановка перед стартовым свистком. kickoffTeam: "player" или "Bot" (кто владеет мячом с центра)
function resetKickoff(kickoffTeam)
	ballPos = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	ballVel = Vector2.new(0, 0)
	playerVel = Vector2.new(0, 0)
	botVel = Vector2.new(0, 0)
	ballRotation = 0
	playerFacingAngle = 0
	botFacingAngle = math.pi
	replayFrames = {}
	replayClock = 0
	replaySampleTimer = 0
	replayActive = false
	replaySkipRequested = false
	if kickoffTeam == "player" then
		playerPos = Vector2.new(425, 300)
		botPos = Vector2.new(790, 300)
	else
		playerPos = Vector2.new(210, 300)
		botPos = Vector2.new(575, 300)
	end
	if cameraMode == "Close" then
		cameraCenter = playerPos * 0.78 + ballPos * 0.22
	else
		cameraCenter = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	end
	botTargetPos = botPos
	botMode = (kickoffTeam == "Bot") and "Attack" or "Challenge"
	botCommitTimer = (kickoffTeam == "Bot") and 2.5 or 0
	botPressTimer = 4
	botRecoveryTimer = 0
	botCornerTimer = 0
	botStuckSampleTimer = 0
	botStuckScore = 0
	botLastSamplePos = botPos
	botLastBallSamplePos = ballPos
	botThinkTimer = 0
	botKickCooldown = 0.6
	ballIdleTimer = 0
	dashTime, dashCooldown = 0, 0
	cancelCharge()
end

function startCountdown()
	state = "Countdown"
	local my = matchToken
	countdownLabel.Visible = true
	task.spawn(function()
		for i = 3, 1, -1 do
			if matchToken ~= my then return end
			countdownLabel.Text = tostring(i)
			playSfx(SFX.Click, 0.08, 0.85 + (3 - i) * 0.18)
			countdownScale.Scale = 1.5
			TweenService:Create(countdownScale, TweenInfo.new(0.45, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Scale = 1 }):Play()
			task.wait(0.65)
		end
		if matchToken ~= my then return end
		countdownLabel.Text = "⚽ ИГРА!"
		playSfx(SFX.Click, 0.10, 1.35)
		countdownScale.Scale = 1.4
		TweenService:Create(countdownScale, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Scale = 1 }):Play()
		state = "Playing"
		task.wait(0.45)
		if matchToken == my then
			countdownLabel.Visible = false
		end
	end)
end

	currentDiff = DIFFICULTIES[selectedDiffKey]
	playerScore, botScore = 0, 0
	shotsPlayer, shotsBot = 0, 0
	timeLeft = SETTINGS.MATCH_TIME
	overtime = false
	keysDown = {}
	cancelCharge()

	diffBadgeLabel.Text = "БОТ: " .. currentDiff.Title
	menuFrame.Visible = false
	gameFrame.Visible = true
	pauseOverlay.Visible = false
	finishOverlay.Visible = false
	goalBanner.Visible = false
	if replayPanel then replayPanel.Visible = false end
	if replaySkipButton then replaySkipButton.Visible = false end

	resetKickoff("player") -- первый мяч с центра — у игрока, честно 😉
	updateHUD()
	startCountdown()
end

function goToMenu()
	matchToken = matchToken + 1
	state = "Menu"
	cancelCharge()
	keysDown = {}
	pauseOverlay.Visible = false
	finishOverlay.Visible = false
	countdownLabel.Visible = false
	goalBanner.Visible = false
	toastLabel.Visible = false
	replayActive = false
	if replayPanel then replayPanel.Visible = false end
	if replaySkipButton then replaySkipButton.Visible = false end

end

local function togglePause(forcePause)
	if state == "Playing" or (forcePause and state == "Countdown") then
		state = "Paused"
		cancelCharge()
		pauseOverlay.Visible = true
	elseif state == "Paused" and not forcePause then
		state = "Playing"
		pauseOverlay.Visible = false
	end
end

function finishMatch()
	state = "Finished"
	cancelCharge()
	countdownLabel.Visible = false
	goalBanner.Visible = false
	if replayPanel then replayPanel.Visible = false end
	if replaySkipButton then replaySkipButton.Visible = false end

	local playerWon = playerScore > botScore
	if playerWon then
		finishTitle.Text = "🏆 ПОБЕДА!"
		finishTitle.TextColor3 = COL.Yellow
	else
		finishTitle.Text = "😤 ПОРАЖЕНИЕ"
		finishTitle.TextColor3 = Color3.fromRGB(255, 110, 100)
	end
	finishScore.Text = tostring(playerScore) .. "  :  " .. tostring(botScore)
	local reason
	if playerScore >= SETTINGS.WIN_GOALS or botScore >= SETTINGS.WIN_GOALS then
		reason = "Матч до " .. tostring(SETTINGS.WIN_GOALS) .. " голов завершён досрочно."
	elseif overtime then
		reason = "Всё решил ЗОЛОТОЙ ГОЛ в дополнительное время!"
	else
		reason = "Время матча вышло."
	end
	finishStats.Text = reason
		.. "\nУдары: ВЫ " .. tostring(shotsPlayer) .. " — " .. tostring(shotsBot) .. " БОТ"
		.. "\nСложность: " .. currentDiff.Title
	finishOverlay.Visible = true
end

-- Запись последних секунд матча для замедленного повтора.
local function recordReplayFrame(dt)
	if state ~= "Playing" and state ~= "Goal" then
		return
	end
	replayClock = replayClock + dt
	replaySampleTimer = replaySampleTimer - dt
	if replaySampleTimer > 0 then
		return
	end
	replaySampleTimer = 1 / 30
	table.insert(replayFrames, {
		t = replayClock,
		p = playerPos,
		b = botPos,
		ball = ballPos,
		rot = ballRotation,
		pf = playerFacingAngle,
		bf = botFacingAngle,
	})
	local cutoff = replayClock - (SETTINGS.REPLAY_SOURCE + 1.6)
	while #replayFrames > 2 and replayFrames[1].t < cutoff do
		table.remove(replayFrames, 1)
	end
end

local function hideReplayUI()
	if replayPanel then
		replayPanel.Visible = false
	end
	if replaySkipButton then
		replaySkipButton.Visible = false
	end
end

local function completeGoalSequence()
	if state ~= "Goal" and state ~= "Replay" then
		return
	end
	replayActive = false
	hideReplayUI()
	goalBanner.Visible = false
	if playerScore >= SETTINGS.WIN_GOALS or botScore >= SETTINGS.WIN_GOALS or overtime then
		finishMatch()
	else
		resetKickoff(lastScorer == "player" and "Bot" or "player")
		startCountdown()
	end
end

local function startGoalReplay()
	if #replayFrames < 2 then
		completeGoalSequence()
		return
	end
	replayLiveState = {
		p = playerPos, b = botPos, ball = ballPos,
		pv = playerVel, bv = botVel, ballv = ballVel,
		rot = ballRotation, pf = playerFacingAngle, bf = botFacingAngle,
	}
	replaySourceEnd = replayFrames[#replayFrames].t
	replaySourceStart = math.max(replayFrames[1].t, replaySourceEnd - SETTINGS.REPLAY_SOURCE)
	replayFrameIndex = 1
	while replayFrameIndex < #replayFrames and replayFrames[replayFrameIndex + 1].t <= replaySourceStart do
		replayFrameIndex = replayFrameIndex + 1
	end
	local sourceDuration = replaySourceEnd - replaySourceStart
	if sourceDuration < 0.2 then
		completeGoalSequence()
		return
	end
	replayDuration = SETTINGS.REPLAY_DURATION -- ровно 7 секунд на экране
	replayTime = 0
	replayActive = true
	state = "Replay"
	goalBanner.Visible = false
	replayPanel.Visible = true
	replaySkipButton.Visible = true
	replayTitle.Text = (lastScorer == "player") and "📺 ПОВТОР ТВОЕГО ГОЛА" or "📺 ПОВТОР ГОЛА БОТА"
	replayProgressFill.Size = UDim2.fromScale(0, 1)
end

local function endGoalReplay()
	if state ~= "Replay" then
		return
	end
	replayActive = false
	hideReplayUI()
	if replayLiveState then
		playerPos = replayLiveState.p
		botPos = replayLiveState.b
		ballPos = replayLiveState.ball
		playerVel = replayLiveState.pv
		botVel = replayLiveState.bv
		ballVel = replayLiveState.ballv
		ballRotation = replayLiveState.rot
		playerFacingAngle = replayLiveState.pf
		botFacingAngle = replayLiveState.bf
	end
	completeGoalSequence()
end

function onGoal(scorer) -- scorer: "player" или "Bot"
	if state ~= "Playing" then
		return
	end
	state = "Goal"
	lastScorer = scorer
	replaySkipRequested = false
	replayActive = false
	cancelCharge()
	if scorer == "player" then
		playerScore = playerScore + 1
	else
		botScore = botScore + 1
	end
	updateHUD()

	goalBanner.Text = (scorer == "player") and "⚽ ГООООЛ!" or "⚽ ГОЛ БОТА..."
	goalBanner.TextColor3 = (scorer == "player") and COL.Yellow or Color3.fromRGB(255, 110, 100)
	goalBanner.Visible = true
	goalBannerScale.Scale = 0.4
	TweenService:Create(goalBannerScale, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Scale = 1 }):Play()
	playSfx(SFX.Applause, (scorer == "player") and 0.26 or 0.15, 1, 3)
	if scorer == "player" then
		spawnConfetti()
	end

	local my = matchToken
	task.delay(0.8, function()
		if matchToken ~= my or state ~= "Goal" then return end
		goalBanner.Visible = false
		if replaySkipRequested then
			completeGoalSequence()
		else
			startGoalReplay()
		end
	end)
end

-- ЧИСТАЯ ФИЗИКА КОНТАКТА (БЕЗ КУРСОРА):
-- Направление удара = линия ЦЕНТР ФИШКИ -> ЦЕНТР МЯЧА в момент касания.
-- Подошёл слева — мяч откинется вправо. Подошёл справа — влево.
-- Подошёл снизу — вверх, сверху — вниз, по диагонали — по той же диагонали.
-- Курсор, старая скорость мяча и боковая скорость фишки направление НЕ меняют.
local function getContactDirection(isBot)
	local chipPos = isBot and botPos or playerPos
	local chipVel = isBot and botVel or playerVel
	local dir = unit2(ballPos - chipPos)
	if dir.Magnitude < 0.01 then
		dir = unit2(chipVel)
	end
	if dir.Magnitude < 0.01 then
		dir = isBot and Vector2.new(-1, 0) or Vector2.new(1, 0)
	end
	return dir
end

-- Удар по мячу только при касании кругов. Возвращает true, если удар произошёл.
local function kickBall(isBot, power01)
	local chipPos = isBot and botPos or playerPos
	local chipVel = isBot and botVel or playerVel
	local toBall = ballPos - chipPos
	local reach = PLAYER_R + BALL_R + (isBot and BOT_KICK_REACH or PLAYER_KICK_REACH)
	if toBall.Magnitude > reach then
		return false
	end

	local dir = getContactDirection(isBot)
	-- Ставим мяч точно на поверхность контакта: никаких случайных перекрытий кругов.
	ballPos = chipPos + dir * (PLAYER_R + BALL_R)

	local p = clamp(power01, 0, 1)
	local speed
	if isBot then
		speed = 240 + p * currentDiff.KickPower
	else
		speed = 280 + p * 760
	end
	-- Только скорость фишки ВДОЛЬ линии контакта усиливает удар; боковая скорость его не уводит.
	local approachSpeed = math.max(0, chipVel:Dot(dir))
	speed = math.min(speed + approachSpeed * 0.35, 1120)
	ballVel = dir * speed

	-- Лёгкая физическая отдача самой фишке
	if isBot then
		botVel = botVel - dir * 18
	else
		playerVel = playerVel - dir * 18
	end

	if isBot then
		shotsBot = shotsBot + 1
	else
		shotsPlayer = shotsPlayer + 1
	end
	playSfx(SFX.Kick, 0.05 + p * 0.12, 0.75 + p * 0.25)
	-- Мяч чуть «пульсирует» от удара
	task.spawn(function()
		ballFrame.Size = fieldSizeToUI(BALL_R * 2 * 1.3, BALL_R * 2 * 1.3)
		task.wait(0.06)
		ballFrame.Size = fieldSizeToUI(BALL_R * 2, BALL_R * 2)
	end)
	return true
end

-- Отпустили ЛКМ / кнопку УДАР
local function releaseKick()
	if not isCharging or state ~= "Playing" then
		cancelCharge()
		return
	end
	local power = math.max(chargePower, 0.25) -- быстрый клик тоже нормально бьёт
	-- Курсор нужен только если мяча рядом нет и вместо удара получится рывок.
	local fallbackAim = chargeAimToGoal and Vector2.new(FIELD_W, GOAL_CENTER_Y) or screenToField(mouseGuiPos())
	isCharging = false
	powerBarBg.Visible = false
	aimLine.Visible = false

	if not kickBall(false, power) then
		-- Мяч не касается фишки — вместо удара делаем рывок к курсору
		if dashCooldown <= 0 then
			local d = unit2(fallbackAim - playerPos)
			if d.Magnitude > 0.01 then
				dashDir = d
				dashTime = 0.18
				dashCooldown = 0.9
			end
		end
	end
	chargePower = 0
	chargeAimToGoal = false
end

--  ИИ БОТА
--------------------------------------------------------------------------------
local function botStep(dt)
	botThinkTimer = botThinkTimer - dt
	botKickCooldown = math.max(0, botKickCooldown - dt)
	botCommitTimer = math.max(0, botCommitTimer - dt)
	botPressTimer = botPressTimer - dt
	if botPressTimer <= 0 then
		-- Даже если игрок долго держит мяч на своей половине,
		-- бот периодически сам идёт в прессинг, а не ждёт у своих ворот.
		botCommitTimer = math.max(botCommitTimer, 2.3)
		botPressTimer = rng:NextNumber(6.5, 9.5)
	end
	local d = currentDiff
	botRecoveryTimer = math.max(0, botRecoveryTimer - dt)

	local function startBotRecovery(fromPos, duration)
		if botRecoveryTimer > 0 then
			return
		end
		botRecoveryTimer = duration or 1.15
		local toCenter = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - fromPos)
		if toCenter.Magnitude < 0.01 then
			toCenter = Vector2.new(1, 0)
		end
		botRecoveryPoint = clampToField(fromPos + toCenter * 190, PLAYER_R)
		botTargetPos = botRecoveryPoint
		botCornerTimer = 0
		botStuckScore = 0
		botThinkTimer = 0
	end

	-- Если мяч почти стоит в углу, а бот его подпирает, бот обязан отойти.
	local stuckBallDist = (ballPos - botPos).Magnitude
	if isInCornerZone(ballPos, 140) and ballVel.Magnitude < 55 and stuckBallDist < 145 then
		botCornerTimer = botCornerTimer + dt
	else
		botCornerTimer = math.max(0, botCornerTimer - dt * 2)
	end
	if botCornerTimer > 0.75 then
		startBotRecovery(ballPos, 1.25)
	end

	-- Общий контроль застревания: бот хочет ехать, но за 0.45 сек почти не сдвинулся.
	botStuckSampleTimer = botStuckSampleTimer + dt
	if botStuckSampleTimer >= 0.45 then
		local botMoved = (botPos - botLastSamplePos).Magnitude
		local ballMoved = (ballPos - botLastBallSamplePos).Magnitude
		local wantsToMove = (botTargetPos - botPos).Magnitude > 38
		local cornerBallSlow = isInCornerZone(ballPos, 140) and ballMoved < 11 and stuckBallDist < 145
		if botRecoveryTimer <= 0 and ((wantsToMove and botMoved < 10) or cornerBallSlow) then
			botStuckScore = botStuckScore + 1
		else
			botStuckScore = math.max(0, botStuckScore - 1)
		end
		if botStuckScore >= 2 then
			startBotRecovery(cornerBallSlow and ballPos or botPos, 1.15)
		end
		botLastSamplePos = botPos
		botLastBallSamplePos = ballPos
		botStuckSampleTimer = 0
	end

	-- «Мозг» бота просыпается раз в Reaction секунд (у сложного — чаще)
	if botThinkTimer <= 0 then
		botThinkTimer = d.Reaction

		-- Предсказание перехвата: где мяч окажется, пока бот до него добежит
		local distToBall = (ballPos - botPos).Magnitude
		local reachTime = clamp(distToBall / math.max(d.Speed, 1), 0, 0.85)
		local predicted = ballPos + ballVel * reachTime * ((0.25 + d.Predict) * 0.78)
		predicted = clampToField(predicted, BALL_R)

		local botDist = (predicted - botPos).Magnitude
		local playerDist = (predicted - playerPos).Magnitude
		local botETA = botDist / math.max(d.Speed, 1)
		local playerETA = playerDist / SETTINGS.PLAYER_SPRINT
		local canWinRace = botETA <= playerETA + (-0.12 + d.Aggression * 0.55)

		local playerHasBall = (ballPos - playerPos).Magnitude <= PLAYER_R + BALL_R + 50
		local botHasBall = (ballPos - botPos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH + 28
		local looseBall = not playerHasBall and not botHasBall
		local ballThreat = (ballPos.X > 720 and ballVel.X > 110)
			or (ballPos.X > 850 and math.abs(ballPos.Y - GOAL_CENTER_Y) < 175)

		-- Линия прессинга: сложный бот встречает игрока очень глубоко.
		-- Проигрывает или мало времени — прессингует ещё выше.
		local pressLine = 470 - d.Aggression * 320
		if botScore < playerScore then
			pressLine = pressLine - 110
		end
		if overtime or (timeLeft < 35 and botScore <= playerScore) then
			pressLine = pressLine - 80
		end

		if ballThreat then
			botMode = "Defend"
		elseif botHasBall or botCommitTimer > 0 or canWinRace
			or predicted.X >= pressLine
			or (looseBall and predicted.X >= pressLine - 110)
			or ballIdleTimer > 1.2 then
			botMode = "Attack"
		elseif playerHasBall and predicted.X >= 260 then
			botMode = "Challenge"
		else
			botMode = "Defend"
		end

		-- Куда бот ХОЧЕТ отправить мяч. Отсюда считается и заход за мяч:
		-- для физического удара бот встаёт С ПРОТИВОПОЛОЖНОЙ стороны.
		local attackTarget
		if isInCornerZone(predicted, 140) then
			-- Сначала вытащить мяч из угла в открытую зону, атака уже потом
			attackTarget = Vector2.new((predicted.X < FIELD_W / 2) and 300 or 700, (predicted.Y < FIELD_H / 2) and 215 or 385)
		elseif ballPos.X > 820 then
			attackTarget = Vector2.new(330, (playerPos.Y < GOAL_CENTER_Y) and 490 or 110)
		elseif predicted.X < d.ShootX then
			local cornerY = (playerPos.Y < GOAL_CENTER_Y) and (GOAL_Y2 - 18) or (GOAL_Y1 + 18)
			attackTarget = Vector2.new(0, cornerY)
		else
			local laneY = predicted.Y * 0.70 + GOAL_CENTER_Y * 0.30 + ((playerPos.Y < predicted.Y) and 38 or -38)
			attackTarget = Vector2.new(predicted.X - 320, clamp(laneY, 70, FIELD_H - 70))
		end
		local shotDir = unit2(attackTarget - predicted)
		local approachPoint = predicted - shotDir * (PLAYER_R + BALL_R + 8)
		local contactDir = unit2(predicted - botPos)
		local alignment = contactDir:Dot(shotDir)

		if botMode == "Attack" or botMode == "Challenge" then
			-- Сначала заходим за мяч с правильной стороны, и только потом идём в мяч.
			-- Поэтому бот не толкает его случайно к своим воротам и реально ведёт атаку влево.
			if botDist > 92 or alignment < 0.20 then
				botTargetPos = approachPoint
			else
				botTargetPos = predicted
			end
		else
			if ballThreat then
				-- Угроза своим воротам: перехватываем мяч, по возможности со стороны ворот.
				if botDist > 80 and alignment < 0.05 then
					botTargetPos = approachPoint
				else
					botTargetPos = predicted
				end
			else
				-- Редкая настоящая оборона — и та высокой линией, а не стоя в воротах.
				local goalPos = Vector2.new(FIELD_W, GOAL_CENTER_Y)
				local dirToBall = unit2(ballPos - goalPos)
				local defPos = goalPos + dirToBall * 315
				botTargetPos = Vector2.new(clamp(defPos.X, 540, 925), clamp(defPos.Y, 85, FIELD_H - 85))
			end
		end
	end

	if botRecoveryTimer > 0 then
		botMode = "Recover"
		botTargetPos = botRecoveryPoint
	end
	botTargetPos = clampToField(botTargetPos, PLAYER_R)

	-- Движение бота: тот же физический подход, что у игрока.
	local toTarget = botTargetPos - botPos
	local targetDist = toTarget.Magnitude
	local speedNow = d.Speed
	if targetDist < 105 then
		speedNow = speedNow * clamp(targetDist / 105, 0.22, 1)
	end
	local ballDistNow = (ballPos - botPos).Magnitude
	if (botMode == "Attack" or botMode == "Challenge") and ballDistNow < 135 then
		-- Рядом с мячом бот контролирует скорость, чтобы не пролетать мяч насквозь
		speedNow = math.min(speedNow, math.max(105, d.Speed * 0.76))
	end
	local desired = Vector2.new(0, 0)
	if targetDist > 4 then
		desired = unit2(toTarget) * speedNow
	end
	local botAccel = d.Accel or 600
	if desired.Magnitude < 1 then
		botAccel = botAccel * 1.15
	end
	botVel = approachVelocity(botVel, desired, botAccel, dt)
	if desired.Magnitude < 1 and botVel.Magnitude < 4 then
		botVel = Vector2.new(0, 0)
	end
	botPos = botPos + botVel * dt
	local clampedBotPos = clampToField(botPos, PLAYER_R)
	-- У борта скорость в стену гаснет, а вдоль борта фишка скользит
	if clampedBotPos.X ~= botPos.X then botVel = Vector2.new(0, botVel.Y) end
	if clampedBotPos.Y ~= botPos.Y then botVel = Vector2.new(botVel.X, 0) end
	botPos = clampedBotPos

	-- Решение об ударе
	if botRecoveryTimer <= 0 and botKickCooldown <= 0 and (ballPos - botPos).Magnitude <= PLAYER_R + BALL_R + BOT_KICK_REACH then
		local aim, power
		if isInCornerZone(ballPos, 140) then
			-- Не добиваем мяч в угол: цель только открытая зона внутри поля
			aim = Vector2.new((ballPos.X < FIELD_W / 2) and 300 or 700, (ballPos.Y < FIELD_H / 2) and 215 or 385)
			power = 0.62
		elseif ballPos.X > 820 then
			-- Опасно у своих ворот — вынос на фланг, подальше от игрока
			aim = Vector2.new(300, (playerPos.Y < GOAL_CENTER_Y) and 490 or 110)
			power = 0.76
		elseif ballPos.X < d.ShootX then
			-- Удар по воротам: в угол, дальний от игрока
			local cornerY = (playerPos.Y < GOAL_CENTER_Y) and (GOAL_Y2 - 18) or (GOAL_Y1 + 18)
			aim = Vector2.new(0, cornerY)
			power = 0.86
		else
			-- Атакующее ведение: толкаем мяч к воротам игрока и ближе к центру
			local laneY = ballPos.Y * 0.70 + GOAL_CENTER_Y * 0.30 + ((playerPos.Y < ballPos.Y) and 38 or -38)
			aim = Vector2.new(ballPos.X - 320, clamp(laneY, 70, FIELD_H - 70))
			power = 0.50
		end

		-- Физический удар нельзя делать с неправильной стороны:
		-- если бот оказался не за мячом, он обежит его, а не ударит назад.
		local desiredDir = unit2(aim - ballPos)
		local contactDirNow = unit2(ballPos - botPos)
		local alignmentNow = contactDirNow:Dot(desiredDir)
		local emergency = ballPos.X > 845 and math.abs(ballPos.Y - GOAL_CENTER_Y) < 180
		local minAlignment = clamp(0.68 - d.AimError * 0.026, 0.02, 0.52) -- сложный бот ждёт более точный заход за мяч
		if isInCornerZone(ballPos, 140) then
			minAlignment = math.max(minAlignment, 0.45)
		end

		if alignmentNow >= minAlignment or (emergency and alignmentNow > -0.35) then
			-- Бот тоже бьёт только по линии контакта своих кругов; неточность даёт сам заход, а не магия.
			if kickBall(true, power) then
				botKickCooldown = d.KickCooldown * rng:NextNumber(0.85, 1.15)
				-- После ведения/выноса бот продолжает атаку за мячом
				botCommitTimer = math.max(botCommitTimer, (ballPos.X > 820) and 1.7 or 2.3)
				botMode = "Attack"
			end
		else
			botTargetPos = ballPos - desiredDir * (PLAYER_R + BALL_R + 10)
			botThinkTimer = math.min(botThinkTimer, 0.05)
		end
	end
end

local function playerStep(dt)
	dashCooldown = math.max(0, dashCooldown - dt)

	local move = Vector2.new(0, 0)
	local touchSpeedScale = 1
	if touchActive then
		local toTouch = touchFieldPos - playerPos
		if toTouch.Magnitude > 12 then
			move = unit2(toTouch)
			-- У пальца не дёргаемся: чем ближе цель, тем мягче скорость
			touchSpeedScale = clamp(toTouch.Magnitude / 95, 0.25, 1)
		end
	elseif not UserInputService:GetFocusedTextBox() then
		if keysDown[Enum.KeyCode.W] or keysDown[Enum.KeyCode.Up] then move = move + Vector2.new(0, -1) end
		if keysDown[Enum.KeyCode.S] or keysDown[Enum.KeyCode.Down] then move = move + Vector2.new(0, 1) end
		if keysDown[Enum.KeyCode.A] or keysDown[Enum.KeyCode.Left] then move = move + Vector2.new(-1, 0) end
		if keysDown[Enum.KeyCode.D] or keysDown[Enum.KeyCode.Right] then move = move + Vector2.new(1, 0) end
	end

	if dashTime > 0 then
		-- Рывок тоже физический: резкий, но не телепорт скорости в один frame
		dashTime = dashTime - dt
		playerVel = approachVelocity(playerVel, dashDir * 500, 1900, dt)
	else
		local speed = SETTINGS.PLAYER_SPEED
		if keysDown[Enum.KeyCode.LeftShift] or keysDown[Enum.KeyCode.RightShift] then
			speed = SETTINGS.PLAYER_SPRINT
		end
		if isCharging then
			speed = speed * 0.72 -- пока целишься — идёшь медленнее
		end
		speed = speed * touchSpeedScale

		local desired = Vector2.new(0, 0)
		if move.Magnitude > 0.01 then
			desired = unit2(move) * speed
		end

		-- Разгон ~0.35 сек, остановка с коротким скольжением, разворот с инерцией
		local acceleration = 820
		if desired.Magnitude < 1 then
			acceleration = 980
		elseif playerVel.Magnitude > 20 and playerVel:Dot(desired) < 0 then
			acceleration = 900
		end
		playerVel = approachVelocity(playerVel, desired, acceleration, dt)
		if desired.Magnitude < 1 and playerVel.Magnitude < 5 then
			playerVel = Vector2.new(0, 0)
		end
	end

	playerPos = playerPos + playerVel * dt
	local clampedPlayerPos = clampToField(playerPos, PLAYER_R)
	-- У борта скорость в стену гаснет, а вдоль борта фишка скользит
	if clampedPlayerPos.X ~= playerPos.X then playerVel = Vector2.new(0, playerVel.Y) end
	if clampedPlayerPos.Y ~= playerPos.Y then playerVel = Vector2.new(playerVel.X, 0) end
	playerPos = clampedPlayerPos
end

-- Фишки не проходят друг сквозь друга
local function resolveChipChip()
	local diff = botPos - playerPos
	local dist = diff.Magnitude
	local minDist = PLAYER_R * 2
	if dist > 0.001 and dist < minDist then
		local n = unit2(diff)
		local push = n * ((minDist - dist) / 2)
		playerPos = playerPos - push
		botPos = botPos + push
		-- Мягкий физический толчок корпусом без дёрганья: выравниваем скорость вдоль линии столкновения
		local closingSpeed = (botVel - playerVel):Dot(n)
		if closingSpeed < 0 then
			playerVel = playerVel + n * (closingSpeed / 2)
			botVel = botVel - n * (closingSpeed / 2)
		end
		playerPos = clampToField(playerPos, PLAYER_R)
		botPos = clampToField(botPos, PLAYER_R)
	end
end

-- Толчок мяча фишкой при ведении (дриблинг)
local function collideBallWithChip(chipPos, chipVel)
	local diff = ballPos - chipPos
	local dist = diff.Magnitude
	local minDist = PLAYER_R + BALL_R
	if dist > 0.001 and dist < minDist then
		local n = diff / dist
		ballPos = chipPos + n * minDist
		local chipSpeedAlong = chipVel:Dot(n)
		local ballSpeedAlong = ballVel:Dot(n)
		if ballSpeedAlong < chipSpeedAlong then
			ballVel = ballVel + n * ((chipSpeedAlong - ballSpeedAlong) * 0.92) -- мяч тяжёлый: ведение не выстреливает им
		end
	end
end

local function inGoalMouth(y)
	return y > GOAL_Y1 and y < GOAL_Y2
end

local function collideBallCorner(sx, sy)
	local dx = (sx > 0) and ballPos.X or (FIELD_W - ballPos.X)
	local dy = (sy > 0) and ballPos.Y or (FIELD_H - ballPos.Y)
	if dx > 190 or dy > 190 then
		return
	end
	local need = CORNER_CUT + BALL_R * math.sqrt(2)
	local sum = dx + dy
	if sum >= need then
		return
	end
	local inward = Vector2.new(sx, sy) / math.sqrt(2)
	ballPos = ballPos + inward * ((need - sum) / math.sqrt(2))
	local vn = ballVel:Dot(inward)
	if vn < 0 then
		local impactSpeed = ballVel.Magnitude
		ballVel = (ballVel - inward * (2 * vn)) * 0.62
		if impactSpeed > 90 then
			playSfx(SFX.Bounce, 0.03, 1.1)
		end
	end
end

local function ballStep(dt)
	-- Два подшага — быстрый мяч не пролетает сквозь фишки и штанги
	for _ = 1, 2 do
		local sdt = dt / 2
		collideBallWithChip(playerPos, playerVel)
		collideBallWithChip(botPos, botVel)

		-- Углы поля скошены и чуть наклонены внутрь: почти остановившийся мяч
		-- сам выкатывается из мёртвого угла, а не остаётся там навсегда.
		if isInCornerZone(ballPos, 115) and ballVel.Magnitude < 110 then
			local inward = unit2(Vector2.new(FIELD_W / 2, FIELD_H / 2) - ballPos)
			ballVel = ballVel + inward * (230 * sdt)
		end

		ballPos = ballPos + ballVel * sdt
		ballVel = ballVel * math.pow(0.34, sdt) -- сильное трение: мяч тяжёлый и быстро останавливается
		if ballVel.Magnitude < 10 then
			ballVel = Vector2.new(0, 0)
		end
		if ballVel.Magnitude > 1100 then
			ballVel = unit2(ballVel) * 1100
		end

		-- Штанги
		for _, post in ipairs(POSTS) do
			local diff = ballPos - post
			local dist = diff.Magnitude
			local minDist = BALL_R + 7
			if dist > 0.001 and dist < minDist then
				local n = diff / dist
				ballPos = post + n * minDist
				local vn = ballVel:Dot(n)
				if vn < 0 then
					ballVel = (ballVel - n * (2 * vn)) * 0.62
					playSfx(SFX.Bounce, 0.04, 1.15)
				end
			end
		end

		-- Верхний и нижний бортики
		if ballPos.Y < BALL_R then
			ballPos = Vector2.new(ballPos.X, BALL_R)
			ballVel = Vector2.new(ballVel.X, math.abs(ballVel.Y) * 0.62)
		elseif ballPos.Y > FIELD_H - BALL_R then
			ballPos = Vector2.new(ballPos.X, FIELD_H - BALL_R)
			ballVel = Vector2.new(ballVel.X, -math.abs(ballVel.Y) * 0.62)
		end

		-- Скошенные углы-отбойники выталкивают мяч обратно в игру
		collideBallCorner(1, 1)
		collideBallCorner(-1, 1)
		collideBallCorner(1, -1)
		collideBallCorner(-1, -1)

		-- Левые ворота (игрока) / левая стенка
		if ballPos.X < BALL_R then
			if inGoalMouth(ballPos.Y) then
				if ballPos.X < -10 then
					onGoal("Bot")
					return
				end
			else
				ballPos = Vector2.new(BALL_R, ballPos.Y)
				ballVel = Vector2.new(math.abs(ballVel.X) * 0.78, ballVel.Y)
			end
		end
		-- Правые ворота (бота) / правая стенка
		if ballPos.X > FIELD_W - BALL_R then
			if inGoalMouth(ballPos.Y) then
				if ballPos.X > FIELD_W + 10 then
					onGoal("player")
					return
				end
			else
				ballPos = Vector2.new(FIELD_W - BALL_R, ballPos.Y)
				ballVel = Vector2.new(-math.abs(ballVel.X) * 0.78, ballVel.Y)
			end
		end
	end
end

local function playingStep(dt)
	playerStep(dt)
	botStep(dt)
	resolveChipChip()
	ballStep(dt)
	if state ~= "Playing" then
		return -- забили гол прямо в ballStep
	end

	-- Застрявший мяч (для логики бота)
	if ballVel.Magnitude < 25
		and (ballPos - playerPos).Magnitude > 120
		and (ballPos - botPos).Magnitude > 120 then
		ballIdleTimer = ballIdleTimer + dt
	else
		ballIdleTimer = 0
	end

	-- Таймер матча 2:00
	if not overtime then
		timeLeft = timeLeft - dt
		if timeLeft <= 0 then
			timeLeft = 0
			if playerScore == botScore then
				overtime = true
				showToast("⚡ ЗОЛОТОЙ ГОЛ! Следующий гол — победный!", 2.5)
			else
				updateHUD()
				finishMatch()
				return
			end
		end
	end
	updateHUD()
end

-- Во время баннера «ГОЛ» мяч просто докатывается в сетке
local function ballSettleStep(dt)
	ballPos = ballPos + ballVel * dt
	ballVel = ballVel * math.pow(0.18, dt)
	ballPos = Vector2.new(
		clamp(ballPos.X, -36, FIELD_W + 36),
		clamp(ballPos.Y, BALL_R, FIELD_H - BALL_R)
	)
end

-- Камера поля: Full помещает ВСЁ большое поле в отведённую зону,
-- Close увеличивает поле в 1.7 раза и плавно следит за игроком (чуть-чуть за мячом).
local function updateFieldCamera(dt)
	if not gameFrame or not fieldFrame then
		return
	end
	local viewport = gameFrame.AbsoluteSize
	if viewport.X < 50 or viewport.Y < 50 then
		return
	end

	local availW = viewport.X * 0.92
	local availH = viewport.Y * 0.82
	local areaCenter = Vector2.new(viewport.X * 0.5, viewport.Y * 0.53)
	local ratio = FIELD_W / FIELD_H
	local fitW, fitH
	if availW / availH > ratio then
		fitH = availH
		fitW = fitH * ratio
	else
		fitW = availW
		fitH = fitW / ratio
	end

	local zoom = 1
	local desired = Vector2.new(FIELD_W / 2, FIELD_H / 2)
	if cameraMode == "Close" then
		zoom = 1.7
		desired = playerPos * 0.78 + ballPos * 0.22
		local visibleW = FIELD_W / zoom
		local visibleH = FIELD_H / zoom
		desired = Vector2.new(
			clamp(desired.X, visibleW / 2 - 38, FIELD_W - visibleW / 2 + 38),
			clamp(desired.Y, visibleH / 2 - 28, FIELD_H - visibleH / 2 + 28)
		)
		if state == "Countdown" then
			cameraCenter = desired
		else
			cameraCenter = cameraCenter:Lerp(desired, clamp(dt * 5.5, 0, 1))
		end
	else
		cameraCenter = desired
	end

	local fieldWpx = fitW * zoom
	local fieldHpx = fitH * zoom
	local topLeft = Vector2.new(
		areaCenter.X - (cameraCenter.X / FIELD_W) * fieldWpx,
		areaCenter.Y - (cameraCenter.Y / FIELD_H) * fieldHpx
	)
	fieldFrame.Position = UDim2.fromOffset(topLeft.X, topLeft.Y)
	fieldFrame.Size = UDim2.fromOffset(fieldWpx, fieldHpx)
end

local function applyDirectionMarkers()
	if playerDirMarker then
		playerDirMarker.Position = UDim2.fromScale(
			0.5 + math.cos(playerFacingAngle) * 0.39,
			0.5 + math.sin(playerFacingAngle) * 0.39
		)
		playerDirMarker.Rotation = math.deg(playerFacingAngle) + 45
	end
	if botDirMarker then
		botDirMarker.Position = UDim2.fromScale(
			0.5 + math.cos(botFacingAngle) * 0.39,
			0.5 + math.sin(botFacingAngle) * 0.39
		)
		botDirMarker.Rotation = math.deg(botFacingAngle) + 45
	end
end

local function updateDirectionMarkers(dt)
	if playerVel.Magnitude > 18 then
		playerFacingAngle = lerpAngle(playerFacingAngle, math.atan(playerVel.Y, playerVel.X), dt * 14)
	end
	if botVel.Magnitude > 18 then
		botFacingAngle = lerpAngle(botFacingAngle, math.atan(botVel.Y, botVel.X), dt * 14)
	end
	applyDirectionMarkers()
end

local function replayStep(dt)
	if not replayActive then
		return
	end
	replayTime = replayTime + dt
	local sourceDuration = replaySourceEnd - replaySourceStart
	local sourceT = replaySourceStart + math.min(sourceDuration, replayTime * (sourceDuration / replayDuration))
	while replayFrameIndex < #replayFrames and replayFrames[replayFrameIndex + 1].t <= sourceT do
		replayFrameIndex = replayFrameIndex + 1
	end
	local a = replayFrames[replayFrameIndex]
	local b = replayFrames[math.min(replayFrameIndex + 1, #replayFrames)]
	local alpha = 0
	if b.t > a.t then
		alpha = clamp((sourceT - a.t) / (b.t - a.t), 0, 1)
	end
	playerPos = a.p:Lerp(b.p, alpha)
	botPos = a.b:Lerp(b.b, alpha)
	ballPos = a.ball:Lerp(b.ball, alpha)
	ballRotation = a.rot + (b.rot - a.rot) * alpha
	playerFacingAngle = lerpAngle(a.pf, b.pf, alpha)
	botFacingAngle = lerpAngle(a.bf, b.bf, alpha)
	applyDirectionMarkers()

	local left = math.max(0, replayDuration - replayTime)
	replayTitle.Text = string.format("%s  •  %.1f c",
		(lastScorer == "player") and "📺 ПОВТОР ТВОЕГО ГОЛА" or "📺 ПОВТОР ГОЛА БОТА", left)
	replayProgressFill.Size = UDim2.fromScale(clamp(replayTime / replayDuration, 0, 1), 1)

	if replayTime >= replayDuration then
		endGoalReplay()
	end
end

local function render()
	playerChip.Position = fieldPosToUI(playerPos)
	botChip.Position = fieldPosToUI(botPos)
	ballFrame.Position = fieldPosToUI(ballPos)
	if ballBody then
		ballBody.Rotation = ballRotation % 360
	end
	applyDirectionMarkers()
end

local function chargeStep(dt)
	if not isCharging then
		return
	end
	chargePower = clamp(chargePower + dt / 0.75, 0, 1)
	powerBarFill.Size = UDim2.fromScale(chargePower, 1)
	if chargePower < 0.5 then
		powerBarFill.BackgroundColor3 = Color3.fromRGB(46, 204, 113):Lerp(COL.Yellow, chargePower * 2)
	else
		powerBarFill.BackgroundColor3 = COL.Yellow:Lerp(Color3.fromRGB(255, 59, 48), (chargePower - 0.5) * 2)
	end

	-- Стрелка показывает ТОЧНУЮ линию центр фишки -> центр мяча.
	local from
	local dir
	local inKickRange = (ballPos - playerPos).Magnitude <= PLAYER_R + BALL_R + PLAYER_KICK_REACH
	if inKickRange then
		from = ballPos
		dir = getContactDirection(false) * 100
	else
		local aim = chargeAimToGoal and Vector2.new(FIELD_W, GOAL_CENTER_Y) or screenToField(mouseGuiPos())
		from = playerPos
		dir = aim - from -- мяч не касается фишки: стрелка показывает только рывок
	end
	if dir.Magnitude > 1 then
		aimLine.Position = fieldPosToUI(from)
		aimLine.Rotation = math.deg(math.atan(dir.Y, dir.X))
		aimLine.Size = UDim2.new((70 + chargePower * 105) / FIELD_W, 0, 0, 5)
		aimLine.Visible = true
	end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if input.UserInputType == Enum.UserInputType.Keyboard then
		keysDown[input.KeyCode] = true
		if input.KeyCode == Enum.KeyCode.P and not gameProcessed then
			if state == "Playing" or state == "Paused" then
				togglePause(false)
			end
		end
		if input.KeyCode == Enum.KeyCode.Space and not gameProcessed then
			if state == "Replay" then
				endGoalReplay()
			elseif state == "Goal" then
				replaySkipRequested = true
			elseif state == "Playing" then
				-- Быстрый физический удар с пробела (бонус во время игры)
				isCharging = true
				chargeAimToGoal = true
				chargePower = 0.55
				releaseKick()
			end
		end
	end

	if input.UserInputType == Enum.UserInputType.MouseButton1 and state == "Playing" and not gameProcessed then
		local mousePos = mouseGuiPos()
		-- Заряжаем удар ТОЛЬКО по клику внутри поля (клики по кнопкам не трогаем)
		if isScreenPosInsideField(mousePos) then
			isCharging = true
			chargeAimToGoal = false
			chargePower = 0.20 -- даже быстрый клик даёт нормальный физический удар
			powerBarBg.Visible = true
		end
	end

	if input.UserInputType == Enum.UserInputType.Touch and state == "Playing" and not gameProcessed then
		local p = toGuiSpace(Vector2.new(input.Position.X, input.Position.Y))
		if isScreenPosInsideField(p) then
			touchActive = true
			touchFieldPos = screenToField(p)
		end
	end
end)
